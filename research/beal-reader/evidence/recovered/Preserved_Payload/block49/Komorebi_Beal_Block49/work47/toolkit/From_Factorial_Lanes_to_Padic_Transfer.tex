\chapter*{Purpose}
\addcontentsline{toc}{chapter}{Purpose}
This book slows one early calculation down until every transformation is visible. The chosen block is the odd-prime standard-window case
\[
p=5,\qquad m=2,\qquad N=mp=10,\qquad j=4,\qquad h=\frac{p-1}{m}=2.
\]
It is large enough to display all ten factorial lanes, nine hypergeometric parameters, a nontrivial reciprocal block, exact polyphase refinement, and prime-power displacement. It is still small enough that every numerator and denominator can be written out exactly.

The same coefficient object is rewritten repeatedly. First it is a residue section of the exponential. Next it is a product of step factorial lanes. Then it becomes a hypergeometric term. After the forced $t^4/4!$ factor is removed, the large factorial quotient collapses to four unit factors. Reciprocal inversion turns that unit structure into a triangular coefficient system. Weighted Mahler transfer then turns the coefficient control into $5$-adic finite-difference control.

No rewrite changes the object. Each rewrite exposes a cancellation that was hidden in the previous coordinates.

\chapter{The one block followed through the whole calculus}
\section{The fixed data}
We work with the residue class $4$ modulo $10$ and the exponential series
\[
e^t=\sum_{r\ge0}\frac{t^r}{r!}.
\]
The transfer projector keeps exactly the terms whose degree is congruent to $4$ modulo $10$:
\[
\SNJ(t)=\BNJ e^t=\sum_{n\ge0}\frac{t^{10n+4}}{(10n+4)!}.
\]
The later reciprocal packet is
\[
\CNJ(t)=\frac{t^4/4!}{\SNJ(t)}
=\sum_{n\ge0}C_n\frac{t^{10n}}{(10n)!}.
\]
The unnormalized congruential row used in the older coefficient tables is
\[
E_n=4!C_n=24C_n.
\]

\section{The dependency chain}
The proof has six distinct layers.
\begin{enumerate}[leftmargin=1.5em]
\item The root-of-unity filter isolates one residue class.
\item The factorial $(10n+4)!$ is split into ten step-$10$ factorial lanes.
\item The lanes are rescaled into Pochhammer factors, and the unique parameter $1$ becomes the ordinary $n!$.
\item Removing the forced $t^4/4!$ packet cancels $(10n)!$ and leaves a four-factor $5$-adic unit.
\item Reciprocal inversion is solved by a lower-triangular recurrence whose diagonal is a $5$-adic unit.
\item The resulting block enters the weighted Mahler transfer, where repeated differences collect powers of $5$.
\end{enumerate}

\chapter{Residue projection before any hypergeometric notation}
\section{Root-of-unity projector}
Let
\[
\omega=e^{2\pi i/10}.
\]
The projector is
\[
(\mathcal B_{10,4}F)(t)
=\frac1{10}\sum_{\ell=0}^{9}\omega^{-4\ell}F(\omega^\ell t).
\]
Substituting $F(t)=e^t$ gives
\[
\frac1{10}\sum_{\ell=0}^{9}\omega^{-4\ell}e^{\omega^\ell t}
=\sum_{r\ge0}\frac{t^r}{r!}
\left(\frac1{10}\sum_{\ell=0}^{9}\omega^{(r-4)\ell}\right).
\]
The finite geometric average is one when $r\equiv4\pmod {10}$ and zero otherwise. Therefore
\[
\boxed{
\SNJ(t)=\sum_{n\ge0}\frac{t^{10n+4}}{(10n+4)!}.
}
\]

\begin{checkpoint}
The first cancellation is spectral. Nine residue classes vanish before any factorial is rewritten.

\chapter{Disassembling one factorial into ten lanes}
\section{Step factorial notation}
For an integer step $N$, define
\[
(a)_{n;N}=\prod_{k=0}^{n-1}(a+Nk),
\qquad
(a)_{0;N}=1.
\]
This is a generalized rising factorial. We call each fixed value of $a$ a factorial lane because it collects one residue class among consecutive factors.

For the chosen block,
\[
\frac{(10n+4)!}{4!}
=5\cdot6\cdots(10n+4).
\]
Group the consecutive factors by their residue inside each block of length ten. This gives
\[
\boxed{
\frac{(10n+4)!}{4!}
=
\prod_{r=5}^{14}(r)_{n;10}.
}
\]
The ten lanes are
\[
(5)_{n;10},(6)_{n;10},\ldots,(14)_{n;10}.
\]
Every integer from $5$ through $10n+4$ appears exactly once.

\chapter{From step factorials to Pochhammer parameters}
\section{Rescaling every lane}
For every lane,
\[
(r)_{n;10}=10^n\left(\frac r{10}\right)_n.
\]
Substituting this into the ten-lane product gives
\[
(10n+4)!
=
4!\,10^{10n}
\prod_{r=5}^{14}
\left(\frac r{10}\right)_n.
\]
Exactly one candidate parameter is equal to one:
\[
\frac{10}{10}=1.
\]
Its factor is
\[
(1)_n=n!.
\]
Move that factor into the standard hypergeometric $n!$ and keep the other nine parameters. The parameter set is
\[
\mathcal A_{10,4}
=
\left\{\frac{1}{2},\ \frac{3}{5},\ \frac{7}{10},\ \frac{4}{5},\ \frac{9}{10},\ \frac{11}{10},\ \frac{6}{5},\ \frac{13}{10},\ \frac{7}{5}\right\}.
\]
Therefore
\[
\boxed{
(10n+4)!
=
4!\,10^{10n}n!
\prod_{\alpha\in\mathcal A_{10,4}}(\alpha)_n.
}
\]

The factor $10(n+1)$ supplies the omitted lane $10n+10$. The other nine factors supply the remaining integers from $10n+5$ through $10n+14$. Thus the ratios agree term by term.

\chapter{The decisive normalization and four-factor collapse}
\section{Remove the forced residue packet}
The section always contains $t^4/4!$. Remove it:
\[
G(t)
=
\frac{4!}{t^4}\SNJ(t).
\]
Write the result in the factorial-normalized block basis
\[
G(t)=\sum_{n\ge0}g_n\frac{t^{10n}}{(10n)!}.
\]
Then
\[
\begin{aligned}
g_n
&=
\frac{4!(10n)!}{(10n+4)!}
\\[4pt]
&=
\frac{24}{(10n+1)(10n+2)(10n+3)(10n+4)}
\\[4pt]
&=
\frac1{\binom{10n+4}{4}}.
\end{aligned}
\]
The original denominator had $(10n+4)!$. After the residue packet and the base factorial are cancelled, only four consecutive factors remain.

\section{Why this is the $5$-adic turning point}
For $r=1,2,3,4$,
\[
10n+r\equiv r\pmod5.
\]
None of the four factors is divisible by $5$. Since $24=4!$ is also a $5$-adic unit,
\[
\boxed{v_5(g_n)=0\quad\text{for every }n\ge0.}
\]
Thus
\[
G(t)\in1+t^{10}\mathbb Z_5[[t^{10}]]
\]
in the factorial-normalized block basis.

\begin{checkpoint}
This is the central numerator-denominator cancellation. The factorial quotient has become a $5$-adic unit because the class $j=4$ ends just before the first forbidden residue $5$.
\end{checkpoint}

\chapter{The standard-window theorem hidden in the example}
\section{General form}
Let $p$ be odd, let $m\mid p-1$, and put
\[
N=mp.
\]
For a residue class satisfying
\[
0\le j<p,
\]
define
\[
d_{N,j}(n)=\binom{Nn+j}{Nn}=\binom{Nn+j}{j}.
\]
Then
\[
d_{N,j}(n)
=
\prod_{r=1}^{j}\frac{Nn+r}{r}.
\]
Because $Nn\equiv0\pmod p$ and $1\le r\le j<p$,
\[
Nn+r\equiv r\not\equiv0\pmod p.
\]
Numerator and denominator have the same zero $p$-adic valuation term by term. Therefore
\[
\boxed{d_{N,j}(n)\in\mathbb Z_p^\times.}
\]

\section{Specialization to the worked block}
For $N=10$ and $j=4$,
\[
d_n=\binom{10n+4}{4}
=\frac{(10n+1)(10n+2)(10n+3)(10n+4)}{24}.
\]
Modulo five,
\[
d_n\equiv\frac{1\cdot2\cdot3\cdot4}{4!}\equiv1\pmod5.
\]
The recursive denominator is not merely integral. It is congruent to one.

\section{Why the boundary matters}
The condition $j<p$ is a clean uniform guarantee. Outside it, a recursive diagonal can acquire factors of $p$. For example,
\[
N=10,\qquad j=5,\qquad n=2
\]
gives
\[
\binom{25}{20}=\binom{25}{5}=53130,
\qquad
v_5(53130)=1.
\]
The general method therefore records the standard window rather than assuming every residue class behaves the same way.

\chapter{Reciprocal inversion and the second cancellation}
\section{The reciprocal block}
Define
\[
\CNJ(t)=G(t)^{-1}
=\sum_{n\ge0}C_n\frac{t^{10n}}{(10n)!}.
\]
Because $G(0)=1$, the inverse exists formally. Multiplying $G(t)\CNJ(t)=1$ and comparing the coefficient of $t^{10n}/(10n)!$ gives
\[
\sum_{q=0}^{n}
\binom{10n}{10q}g_{n-q}C_q
=\delta_{n0}.
\]
For $n\ge1$,
\[
\boxed{
C_n
=-\sum_{q=0}^{n-1}
\binom{10n}{10q}g_{n-q}C_q.
}
\]
Every kernel coefficient is in $\mathbb Z_5$ and the diagonal coefficient is one. Induction gives
\[
\boxed{C_n\in\mathbb Z_5\quad\text{for every }n.}
\]
Since $E_n=24C_n$ and $24$ is a unit,
\[
E_n\in\mathbb Z_5.
\]

\section{The same recurrence in the older triangular form}
Multiplying the original section by the reciprocal gives
\[
\sum_{q=0}^{n}\binom{10n+4}{10q}C_q=\delta_{n0}.
\]
The diagonal term is
\[
d_n=\binom{10n+4}{10n}.
\]
The bridge between the two forms is the exact identity
\[
\boxed{
\frac{\binom{10n+4}{10q}}{d_n}
=
\binom{10n}{10q}g_{n-q}.
}
\]
Proof:
\[
\begin{aligned}
\frac{\binom{10n+4}{10q}}{\binom{10n+4}{10n}}
&=
\frac{4!(10n)!}{(10q)!(10(n-q)+4)!}
\\[4pt]
&=
\binom{10n}{10q}
\frac{4!(10(n-q))!}{(10(n-q)+4)!}
\\[4pt]
&=
\binom{10n}{10q}g_{n-q}.
\end{aligned}
\]
The large diagonal denominator cancels uniformly from every numerator term.

\begin{checkpoint}
The third cancellation is triangular. Instead of dividing one large rational numerator at the end, divide the common diagonal from every term first. The resulting kernel is visibly $5$-adically integral.
\end{checkpoint}

\section{Refinement from modulus ten to modulus fifty}
Every integer congruent to $4$ modulo $10$ has exactly one of the five residues
\[
4,14,24,34,44\pmod{50}.

Therefore the projectors satisfy
\[
\boxed{
\mathcal B_{10,4}
=
\mathcal B_{50,4}
+
\mathcal B_{50,14}
+
\mathcal B_{50,24}
+
\mathcal B_{50,34}
+
\mathcal B_{50,44}.
}
\]
Applied to $e^t$,
\[
\SNJ(t)
=
\sum_{r\in\{4,14,24,34,44\}}
\mathcal B_{50,r}e^t.
\]
Each child has its own hypergeometric presentation with forty-nine lower parameters. Their supports are disjoint. Their sum reconstructs the parent exactly.

\section{What refinement does to the p-adic proof}
The coarse class $j=4$ lies in the standard $5$-adic window. The children do not all lie in a standard window when written as classes modulo fifty. The safe arithmetic statement is therefore attached to the complete parent packet, while the child packets provide a finer analytic or computational atlas.

The distinction matters. Exact refinement preserves the object. It does not guarantee that every child separately inherits the same simplest denominator proof. When the children are synthesized, the parent unit cancellation returns.

\[

\]
The relevant finite-difference operator is
\[
\Delta_2f(n)=f(n+2)-f(n).
\]
Repeated and mixed differences are formed by composition.

\section{The weighted transfer statement}
Under the weighted Amice--Mahler hypotheses for the congruential Euler block, the mixed difference bound is
\[
\boxed{
 v_5\!\left(
 \Delta_{2q_1}\cdots\Delta_{2q_s}E_n
 \right)
 \ge
 \min\left
 \{2n,
 s+\sum_{\nu=1}^{s}v_5(q_\nu)
 \right\}.
}
\]
The two terms describe different saturation mechanisms.
\begin{itemize}[leftmargin=1.5em]
\item The factor $2n$ comes from the weighted depth already present in the block coefficient.
\item Every ordinary difference contributes one factor of five.
\item Replacing a unit displacement by a multiple of $5^r$ contributes $r$ additional factors.
\end{itemize}

\section{Why the earlier rewrites were necessary}
The hypergeometric formula alone does not prove the finite-difference law. It supplies the exact factorial structure. The four-factor normalization proves that the recursive diagonal is a unit. The finite-difference estimate is the end of this chain, not an isolated congruence.

\chapter{The complete cancellation chain}
The worked example has five cancellations, each occurring in a different coordinate system.

\section{Cancellation one: residue support}
The root filter removes nine of ten Taylor residue classes.

\section{Cancellation two: factorial lanes}
The product
\[
(10n+4)!/4!
\]
is partitioned into ten step-$10$ factorials. Rescaling the lanes introduces rational Pochhammer parameters, but the common factor $10^{10n}$ clears every new denominator.

\section{Cancellation three: the hidden $n!$}
The candidate parameter $10/10=1$ gives $(1)_n=n!$. It is omitted from the displayed lower-parameter list because the hypergeometric definition already contains it.

\section{Cancellation four: the standard-window tail}
Removing $t^4/4!$ and cancelling $(10n)!$ reduces the factorial quotient to
\[
\frac{24}{(10n+1)(10n+2)(10n+3)(10n+4)}.
\]
Every denominator factor is a $5$-adic unit.

\section{Cancellation five: the reciprocal diagonal}
The factor
\[
\binom{10n+4}{4}
\]
is common to every term in the triangular numerator. Dividing it termwise converts the recurrence to one whose kernels are binomial integers times $5$-adic units.

\section{Transfer into differences}
The weighted Mahler coordinates convert those coefficient bounds into repeated difference bounds. Polyphase refinement provides finer child packets without changing the reconstructed parent.

\begin{theorem}[General standard-window mechanism]
Let $p$ be odd, let $m\mid p-1$, put $N=mp$, and take $0\le j<p$. Then the hypergeometric residue section admits the factorial-lane decomposition
\[
(Nn+j)!=j!N^{Nn}n!\prod_{\alpha\in\mathcal A_{N,j}}(\alpha)_n,
\]
the normalized tail
\[
\frac{j!(Nn)!}{(Nn+j)!}
=
\binom{Nn+j}{j}^{-1}
\]
is a $p$-adic unit, and reciprocal inversion can be performed without diagonal $p$-adic loss. Under the weighted Amice--Mahler hypotheses, the resulting block satisfies the mixed Kummer finite-difference bounds.
\end{theorem}

\begin{boundary}
The factorial and reciprocal identities are exact algebraic statements. The mixed Kummer theorem additionally uses the weighted Amice--Mahler hypotheses. The finite ledgers verify the chosen example and display the surplus. They do not replace the general analytic hypothesis.
\end{boundary}
