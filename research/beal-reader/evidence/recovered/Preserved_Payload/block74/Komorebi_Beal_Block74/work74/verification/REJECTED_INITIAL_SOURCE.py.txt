#!/usr/bin/env python3
"""Exact finite checks for the Block74 trace probe. Standard library only."""
from __future__ import annotations
import argparse,hashlib,itertools,json
from pathlib import Path

class Quad:
    """Z/mod[w]/(w^2-5), as pairs; no floating arithmetic."""
    def __init__(self,mod): self.mod=mod
    def z(self,n):return (n%self.mod,0)
    def add(self,x,y):return ((x[0]+y[0])%self.mod,(x[1]+y[1])%self.mod)
    def neg(self,x):return (-x[0]%self.mod,-x[1]%self.mod)
    def sub(self,x,y):return self.add(x,self.neg(y))
    def mul(self,x,y):return ((x[0]*y[0]+5*x[1]*y[1])%self.mod,(x[0]*y[1]+x[1]*y[0])%self.mod)
    def inv(self,x):
        n=(x[0]*x[0]-5*x[1]*x[1])%self.mod
        if n%7==0:raise ValueError('nonunit')
        t=pow(n,-1,self.mod);return (x[0]*t%self.mod,-x[1]*t%self.mod)
    def power(self,x,e):
        y=self.z(1)
        while e:
            if e&1:y=self.mul(y,x)
            x=self.mul(x,x);e//=2
        return y
    def scale(self,n,x):return self.mul(self.z(n),x)
    def mat(self,a,b,c,d):return (a,b,c,d)
    def mm(self,A,B):
        return tuple(self.add(self.mul(A[2*i],B[j]),self.mul(A[2*i+1],B[2+j])) for i in range(2) for j in range(2))
    def mi(self,A):
        u=self.inv(self.sub(self.mul(A[0],A[3]),self.mul(A[1],A[2])))
        return tuple(self.mul(u,x) for x in (A[3],self.neg(A[1]),self.neg(A[2]),A[0]))
    def tr(self,A):return self.add(A[0],A[3])
    def det(self,A):return self.sub(self.mul(A[0],A[3]),self.mul(A[1],A[2]))
    def ident(self):return (self.z(1),self.z(0),self.z(0),self.z(1))

def val(x,j):
    if x==(0,0):return j
    t=0
    while x[0]%7==0 and x[1]%7==0:
        x=(x[0]//7,x[1]//7);t+=1
    return t

def compute(contract,patterns):
    digest=hashlib.sha256();count=nonrational=nonflat=0;hist={};examples=[]
    choices=itertools.product(contract['depths'],contract['valuation_exponents'],contract['c_units'],contract['natural_v'],contract['orientations'],contract['variations'])
    for n,s,cu,v,eps,i in choices:
        c=7**s*cu;t=s+1;j=n+t+1;R=Quad(7**j)
        kappa=v*pow(2*eps,-1,7)%7
        # Actual local toric line is congruent to the natural line modulo 7^n.
        a=R.add(R.z(1),R.scale(7,(i+cu,v+i)))
        b=((2*i+v)%R.mod,(cu+1)%R.mod)
        d=R.scale(7**n,(i+1,cu+i))
        e=R.add(R.z(pow(kappa,-1,7)),R.scale(7,(cu+v,2*i+1)))
        P=(a,b,d,e)
        E=(R.z(0),R.z(1),R.z(0),R.z(0))
        J=R.mm(R.mm(P,E),R.mi(P))
        assert R.mm(J,J)==(R.z(0),)*4
        assert val(J[0],j)>=n and val(J[2],j)>=2*n
        assert J[1][0]%7==kappa and J[1][1]%7==0
        U=tuple(R.add(x,R.scale(7*c,y)) for x,y in zip(R.ident(),J))
        beta_code=1+((i+7*cu+5*v+3*n+11*s)%48)
        beta=(beta_code%7,beta_code//7)
        x=R.add(R.z(1),R.scale(7,(i+3,v+cu)))
        y=R.scale(7,(cu+i,2*v+i))
        z=R.scale(7**n,beta)
        Q=1+7*(1+i+v+cu+n)
        w=R.mul(R.add(R.z(Q),R.mul(y,z)),R.inv(x))
        V=(x,y,z,w)
        assert R.det(V)==R.z(Q) and R.det(U)==R.z(1)
        delta=R.add(R.sub(R.sub(R.tr(R.mm(U,V)),R.tr(U)),R.tr(V)),R.z(2))
        direct=R.tr(R.mm(tuple(R.sub(x,y) for x,y in zip(U,R.ident())),tuple(R.sub(x,y) for x,y in zip(V,R.ident()))))
        assert delta==direct
        assert val(delta,j)==n+t
        lead=(delta[0]//7**(n+t)%7,delta[1]//7**(n+t)%7)
        expected=(cu*kappa*beta[0]%7,cu*kappa*beta[1]%7)
        assert lead==expected
        # A higher-order toric diagonal term really occurs, and is not dropped blindly.
        nonflat+=int(U[0]!=R.z(1) or U[3]!=R.z(1))
        nonrational+=int(any(q[1] for q in U+V))
        item={'n':n,'c':c,'v':v,'epsilon':eps,'i':i,'precision':j,'beta':beta,'mixed_trace':delta,'valuation':n+t,'leading':lead}
        digest.update((json.dumps(item,sort_keys=True,separators=(',',':'))+'\n').encode())
        if i==0 and cu==1 and v==1 and eps==1:examples.append(item)
        count+=1;hist[str(n)]=hist.get(str(n),0)+1
    # Residual trace pairing: all possible nonzero detecting beta values.
    residue_digest=hashlib.sha256();residue_count=0
    R=Quad(343);k=Quad(7)
    for nc,bc in itertools.product(range(49),range(1,49)):
        nu=(nc%7,nc//7);be=(bc%7,bc//7)
        U=(R.z(1),R.scale(7,nu),R.z(0),R.z(1))
        V=(R.z(1),R.z(0),R.scale(7,be),R.z(1))
        z=R.sub(R.tr(R.mm(U,V)),R.z(2));digit=(z[0]//49%7,z[1]//49%7)
        assert k.mul(digit,k.inv(be))==nu
        residue_digest.update(bytes([nc,bc,*digit]));residue_count+=1
    # Primitive local unit coordinates. Residue torsion order is 48, prime to 7.
    R=Quad(49);phi=(25,25);three=(3,0)
    images=[]
    for x in (phi,three):
        y=R.power(x,48);images.append(((y[0]-1)//7%7,y[1]//7%7))
    assert images==[(0,5),(6,0)]
    counts=[0]*49;units=0
    for a,b in itertools.product(range(49),repeat=2):
        if a%7==b%7==0:continue
        u=R.power((a,b),48);assert (u[0]-1)%7==0 and u[1]%7==0
        x=((u[0]-1)//7)%7;y=(u[1]//7)%7
        counts[x+7*y]+=1;units+=1
    assert units==2304 and set(counts)=={48}
    # Finite model of the cyclotomic restriction argument: commutative C6 x C7^s.
    pairs=surv=0
    for a,b in itertools.product(range(49),repeat=2):
        av=(a%7,a//7);bv=(b%7,b//7)
        # Cocycle compatibility with commuting generators, weights 3 and 1.
        lhs=((2*bv[0])%7,(2*bv[1])%7)
        if lhs==(0,0):
            assert bv==(0,0);surv+=1
            z=((4*av[0])%7,(4*av[1])%7) # inverse of 2 modulo 7
            assert ((2*z[0])%7,(2*z[1])%7)==av
        pairs+=1
    # Joint-image projection gives a g with residual matrix I and beta nonzero.
    eigencases=0
    for a,b in itertools.product(range(49),repeat=2):
        av=(a%7,a//7);bv=(b%7,b//7)
        pa=((3*av[0]-3*av[0])*4%7,(3*av[1]-3*av[1])*4%7)
        pb=((5*bv[0]-3*bv[0])*4%7,(5*bv[1]-3*bv[1])*4%7)
        assert pa==(0,0) and pb==bv;eigencases+=1
    controls={
      'same_individual_traces':{'U1_trace':2,'U2_trace':2,'V_trace':2,'U1V_trace_mod343':51,'U2V_trace_mod343':100,'determinants':[1,1,1]},
      'contaminated_probe':{'predicted_if_uniformizer_only':1,'actual_mixed_digit':3,'reason':'nonzero diagonal first digit contaminates the pairing'},
      'wrong_depth_denominator':{'n':2,'c':1,'delta':343,'mod343':0,'mod2401':343},
      'coefficient_trace_loss':{'nonzero_digit':[0,1],'trace_to_F7':0},
      'beta_zero':{'reconstruction_allowed':False},
      'not_arbitrary_primes':{'joint_word_labels_required':True}}
    # Verify the displayed concrete controls by matrix multiplication.
    R=Quad(343);I=R.ident();V=(R.z(1),R.z(0),R.z(7),R.z(1))
    for nu,expected in ((1,51),(2,100)):
        U=(R.z(1),R.z(7*nu),R.z(0),R.z(1));assert R.tr(R.mm(U,V))==R.z(expected)
    U=(R.z(8),R.z(7),R.z(0),R.inv(R.z(8)))
    V=(R.z(8),R.z(0),R.z(7),R.inv(R.z(8)))
    d=R.add(R.sub(R.sub(R.tr(R.mm(U,V)),R.tr(U)),R.tr(V)),R.z(2))
    assert d==R.z(147)
    # The eight frozen patterns remain. A trace formula supplies no new numerical trace.
    tuples=patterns['survivors'];assert len(tuples)==8
    return {'schema':'beal.block74.trace-probe.v1',
       'matrix_family':{'cases':count,'nonrational':nonrational,'nontrivial_toric_diagonals':nonflat,'depth_histogram':hist,'ordered_sha256':digest.hexdigest(),'examples':examples},
       'residue_reconstruction':{'cases':residue_count,'ordered_sha256':residue_digest.hexdigest()},
       'unit_basis':{'phi_image':images[0],'three_image':images[1],'unit_residues_checked':units,'image_multiplicities':counts},
       'cyclotomic_cocycle_model':{'generator_value_pairs':pairs,'cocycles':surv,'coboundaries':surv,'H1_dimension':0},
       'joint_projection':{'vectors':eigencases,'weights':[3,5],'inverse_difference':4},
       'controls':controls,
       'status':{'patterns_before':8,'patterns_after':8,'patterns_eliminated':0,'actual_joint_word_traces_computed':False,'Frobenius_point_counts':0,'Lean_compilation':False,'independent_mathematical_review':False}}

def main():
    ap=argparse.ArgumentParser();ap.add_argument('--output',type=Path,required=True);args=ap.parse_args()
    root=Path(__file__).resolve().parents[2]
    result=compute(json.loads((root/'work74/inputs/TEST_CONTRACT.json').read_text()),json.loads((root/'work74/inputs/B72_B71_B70_refinement70.json').read_text()))
    args.output.parent.mkdir(parents=True,exist_ok=True);args.output.write_text(json.dumps(result,sort_keys=True,indent=2)+'\n')
    print('PASS_FINITE_TRACE_PROBE',result['matrix_family']['cases'],'matrix cases')
if __name__=='__main__':main()
