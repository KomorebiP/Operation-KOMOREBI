\section{Kummer Globalization and the Fermat Character Packet}

\subsection{The contact modulus}
Let
\[
 c=\prod_{i=1}^{r}p_i^{e_i},
 \qquad
 R_c=\prod_i p_i,
 \qquad
 M=\frac{c}{R_c}
 =\prod_i p_i^{e_i-1}.
\]
The filtered output contact packet is the cyclic group
\[
 \Z/M\Z.
\]
The first global question is whether this packet can be realized by one arithmetic class without losing most of its order.

\subsection{The global Kummer map}
Kummer theory gives
\[
 H^1(\Q,\mu_M)
 \cong
 \Q^\times/(\Q^\times)^M.
\]
Define
\[
 \boxed{
 \kappa_c:
 \Z/M\Z
 \longrightarrow
 H^1(\Q,\mu_M),
 \qquad
 k\longmapsto[c^k].
 }
\]
The valuation vector of $c$ is
\[
 (e_1,\ldots,e_r)
 \in
 (\Z/M\Z)^r.
\]

\begin{theorem}[Exact Kummer kernel]
Let
\[
 g_c=\gcd(M,e_1,\ldots,e_r).
\]
Then
\[
 \boxed{
 |\ker\kappa_c|=g_c,
 \qquad
 |\im\kappa_c|=M/g_c.
 }
\]
\end{theorem}

\begin{proof}
The class $[c]^k$ is trivial exactly when every prime valuation $ke_i$ is divisible by $M$. The least positive such $k$ is
\[
 \operatorname{lcm}_i
 \frac{M}{\gcd(M,e_i)}
 =
 \frac{M}{\gcd(M,e_1,\ldots,e_r)}.
\]
This is the order of the image. The kernel size follows.
\end{proof}

Because
\[
 g_c\le\min_i e_i\le\log_2 c,
\]
we obtain the universal near-injectivity estimate
\[
 \boxed{
 |\ker\kappa_c|\le\log_2c.
 }
\]
This closes the global-occupancy problem for the contact packet up to logarithmic loss.

\subsection{Balanced Kummer class}
The class of $c$ does not remember the additive relation. A better coordinate is
\[
 \beta=-\frac ba=1-\frac ca.
\]
Let
\[
 \nu_q=v_q(b)-v_q(a)
\]
for input primes $q\mid ab$. The order of $[\beta]$ in
\[
 \Q^\times/(\Q^\times)^M
\]
is determined by the signed valuation vector, with one possible dyadic sign correction when $M$ is even. If
\[
 g_{\mathrm{val}}
 =
 \gcd(M,\{\nu_q\}),
 \qquad
 m_0=M/g_{\mathrm{val}},
\]
then the order is $m_0$ unless $M$ is even and $m_0$ is odd, in which case it is $2m_0$.

The balanced class retains the input-output relation and usually has full contact order. Its kernel remains logarithmically bounded.

\subsection{The Kummer--Fermat lift}
Choose algebraic numbers $u$ and $v$ satisfying
\[
 u^M=\frac ac,
 \qquad
 v^M=\frac bc.
\]
Then
\[
 \boxed{u^M+v^M=1.}
\]
Thus
\[
 P_M=[u:v:1]
\]
lies on the Fermat curve
\[
 F_M:X^M+Y^M=Z^M.
\]
The construction is canonical up to the action of $\mu_M^2$.

\begin{theorem}[Exact height compression]
\[
 \boxed{
 h_{\mathcal O(1)}(P_M)=\frac{\log c}{M}.
 }
\]
\end{theorem}

\begin{proof}
The coordinatewise $M$th-power map sends $P_M$ to $[a:b:c]$. Absolute logarithmic projective height multiplies by $M$ under this map. Since the primitive positive triple has height $\log c$, the result follows.
\end{proof}

The contact modulus divides the displayed coordinate height by the full powerful part.

\subsection{Fermat character polyphase}
The group $\mu_M^2$ acts by
\[
 (\xi,\eta)\cdot(X,Y)=(\xi X,\eta Y).
\]
Let $\sigma$ and $\tau$ denote the two coordinate rotations. For
\[
 (r,s)\in(\Z/M\Z)^2,
\]
define
\[
 P_{r,s}
 =
 \frac1{M^2}
 \sum_{i,j=0}^{M-1}
 \zeta_M^{-ri-sj}
 \sigma^i\tau^j.
\]
Character orthogonality gives
\[
 P_{r,s}P_{r',s'}
 =
 \delta_{r,r'}\delta_{s,s'}P_{r,s}
\]
and
\[
 \boxed{
 \sum_{r,s}P_{r,s}=I.
 }
\]
This is an exact two-dimensional character filter bank.

\subsection{Line-bundle cost of the full Fermat model}
For $M\ge4$, adjunction gives
\[
 K_{F_M}\cong\mathcal O_{F_M}(M-3).
\]
For compatible Weil-height normalizations,
\[
 h_{K_{F_M}}(P_M)
 =
 (M-3)h_{\mathcal O(1)}(P_M)+O(1)
 =
 \frac{M-3}{M}\log c+O(1).
\]
If the metric on $K_{F_M}$ is defined by the induced tensor-power metric on $\mathcal O(1)^{\otimes(M-3)}$, the displayed equality holds without the $O(1)$ term. Thus the full plane-curve polarization reintroduces a factor of order $M$ into the height accounting.

\begin{proposition}[Line-bundle cost of the full plane model]
The Kummer lift reduces the $\mathcal O(1)$ coordinate height by $M$, while the canonical line bundle of the complete Fermat curve has degree factor $M-3$. This rules out a height saving obtained merely by retaining the full plane-curve carrier with its induced polarization. It does not rule out a lower-complexity quotient equipped with a different arithmetic metric.
\end{proposition}

A successful route must therefore select and control a much smaller quotient of the character bank. Fermat character decompositions and Jacobi-sum factors provide the established ambient framework for such quotients~\cite{AokiFermat,OtsuboYamazaki,ColemanMcCallum}.
