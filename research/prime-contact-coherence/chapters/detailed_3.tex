\PartChapter{Detailed Theory: Kummer, Fermat, Polylogarithmic, and Coherence Objects}{Kummer Globalization and the Fermat Character Packet}

\section{The contact modulus}
Let
\[
 c=\prod_{i=1}^{r}p_i^{e_i},
 \qquad
 R_c=\prod_i p_i,
 \qquad
 M=\frac{c}{R_c}
 =\prod_i p_i^{e_i-1}.
\]
The filtered output contact packet is the cyclic group
\[
 \Z/M\Z.
\]
The first global question is whether this packet can be realized by one arithmetic class without losing most of its order.

\section{The global Kummer map}
Kummer theory gives
\[
 H^1(\Q,\mu_M)
 \cong
 \Q^\times/(\Q^\times)^M.
\]
Define
\[
 \boxed{
 \kappa_c:
 \Z/M\Z
 \longrightarrow
 H^1(\Q,\mu_M),
 \qquad
 k\longmapsto[c^k].
 }
\]
The valuation vector of $c$ is
\[
 (e_1,\ldots,e_r)
 \in
 (\Z/M\Z)^r.
\]

\begin{theorem}[Exact Kummer kernel]
Let
\[
 g_c=\gcd(M,e_1,\ldots,e_r).
\]
Then
\[
 \boxed{
 |\ker\kappa_c|=g_c,
 \qquad
 |\im\kappa_c|=M/g_c.
 }
\]
\end{theorem}

\begin{proof}
The class $[c]^k$ is trivial exactly when every prime valuation $ke_i$ is divisible by $M$. The least positive such $k$ is
\[
 \operatorname{lcm}_i
 \frac{M}{\gcd(M,e_i)}
 =
 \frac{M}{\gcd(M,e_1,\ldots,e_r)}.
\]
This is the order of the image. The kernel size follows.
\end{proof}

Because
\[
 g_c\le\min_i e_i\le\log_2 c,
\]
we obtain the universal near-injectivity estimate
\[
 \boxed{
 |\ker\kappa_c|\le\log_2c.
 }
\]
This closes the global-occupancy problem for the contact packet up to logarithmic loss.

\section{Balanced Kummer class}
The class of $c$ does not remember the additive relation. A better coordinate is
\[
 \beta=-\frac ba=1-\frac ca.
\]
Let
\[
 \nu_q=v_q(b)-v_q(a)
\]
for input primes $q\mid ab$. The order of $[\beta]$ in
\[
 \Q^\times/(\Q^\times)^M
\]
is determined by the signed valuation vector, with one possible dyadic sign correction when $M$ is even. If
\[
 g_{\mathrm{val}}
 =
 \gcd(M,\{\nu_q\}),
 \qquad
 m_0=M/g_{\mathrm{val}},
\]
then the order is $m_0$ unless $M$ is even and $m_0$ is odd, in which case it is $2m_0$.

The balanced class retains the input-output relation and usually has full contact order. Its kernel remains logarithmically bounded.

\section{The Kummer--Fermat lift}
Choose algebraic numbers $u$ and $v$ satisfying
\[
 u^M=\frac ac,
 \qquad
 v^M=\frac bc.
\]
Then
\[
 \boxed{u^M+v^M=1.}
\]
Thus
\[
 P_M=[u:v:1]
\]
lies on the Fermat curve
\[
 F_M:X^M+Y^M=Z^M.
\]
The construction is canonical up to the action of $\mu_M^2$.

\begin{theorem}[Exact height compression]
\[
 \boxed{
 h_{\mathcal O(1)}(P_M)=\frac{\log c}{M}.
 }
\]
\end{theorem}

\begin{proof}
The coordinatewise $M$th-power map sends $P_M$ to $[a:b:c]$. Absolute logarithmic projective height multiplies by $M$ under this map. Since the primitive positive triple has height $\log c$, the result follows.
\end{proof}

The contact modulus divides the displayed coordinate height by the full powerful part.

\section{Fermat character polyphase}
The group $\mu_M^2$ acts by
\[
 (\xi,\eta)\cdot(X,Y)=(\xi X,\eta Y).
\]
Let $\sigma$ and $\tau$ denote the two coordinate rotations. For
\[
 (r,s)\in(\Z/M\Z)^2,
\]
define
\[
 P_{r,s}
 =
 \frac1{M^2}
 \sum_{i,j=0}^{M-1}
 \zeta_M^{-ri-sj}
 \sigma^i\tau^j.
\]
Character orthogonality gives
\[
 P_{r,s}P_{r',s'}
 =
 \delta_{r,r'}\delta_{s,s'}P_{r,s}
\]
and
\[
 \boxed{
 \sum_{r,s}P_{r,s}=I.
 }
\]
This is an exact two-dimensional character filter bank.

\section{Line-bundle cost of the full Fermat model}
For $M\ge4$, adjunction gives
\[
 K_{F_M}\cong\mathcal O_{F_M}(M-3).
\]
For compatible Weil-height normalizations,
\[
 h_{K_{F_M}}(P_M)
 =
 (M-3)h_{\mathcal O(1)}(P_M)+O(1)
 =
 \frac{M-3}{M}\log c+O(1).
\]
If the metric on $K_{F_M}$ is defined by the induced tensor-power metric on $\mathcal O(1)^{\otimes(M-3)}$, the displayed equality holds without the $O(1)$ term. Thus the full plane-curve polarization reintroduces a factor of order $M$ into the height accounting.

\begin{proposition}[Line-bundle cost of the full plane model]
The Kummer lift reduces the $\mathcal O(1)$ coordinate height by $M$, while the canonical line bundle of the complete Fermat curve has degree factor $M-3$. This rules out a height saving obtained merely by retaining the full plane-curve carrier with its induced polarization. It does not rule out a lower-complexity quotient equipped with a different arithmetic metric.
\end{proposition}

A successful route must therefore select and control a much smaller quotient of the character bank. Fermat character decompositions and Jacobi-sum factors provide the established ambient framework for such quotients~\cite{AokiFermat,OtsuboYamazaki,ColemanMcCallum}.

\chapter{The Polylogarithmic Three-Point Path Packet}

\section{Root-distribution compression}
Let $u^M=z$ and let $\zeta_M$ be a primitive $M$th root of unity. For $n\ge1$,
\[
 \boxed{
 \sum_{j=0}^{M-1}
 \operatorname{Li}_n(\zeta_M^ju)
 =
 M^{1-n}\operatorname{Li}_n(u^M).
 }
\]
Inside the disk of convergence, this follows by summing the defining series and using the root-of-unity filter. Analytic continuation extends the identity with compatible branches.

The inverse synthesis gain at weight $n$ is $M^{n-1}$. Through weights one, two, and three, the total gain is
\[
 \boxed{1\cdot M\cdot M^2=M^3.}
\]
The analytic branch count becomes fixed, but the integral transition matrix retains a large determinant.

\section{Three-point functional closure}
Set
\[
 z=\frac ac,
 \qquad
 1-z=\frac bc,
 \qquad
 \beta=1-\frac1z=-\frac ba.
\]
The Rogers dilogarithm
\[
 L_2(z)
 =
 \operatorname{Li}_2(z)
 +
 \frac12\log z\log(1-z)
\]
satisfies
\[
 \boxed{
 L_2(z)+L_2(1-z)=\frac{\pi^2}{6}.
 }
\]
At weight three,
\[
 \boxed{
 \begin{aligned}
 &\operatorname{Li}_3(z)
 +\operatorname{Li}_3(1-z)
 +\operatorname{Li}_3(\beta)\\
 &\qquad=
 \zeta(3)
 +\frac16\log^3z
 +\frac{\pi^2}{6}\log z
 -\frac12\log^2z\log(1-z).
 \end{aligned}
 }
\]
The balanced Kummer coordinate is therefore exactly the third argument in the depth-three functional closure.

A fixed path packet is
\[
 \Pi_{\le3}(z)
 =
 \left(
 \log z,
 \log(1-z),
 L_2(z),
 \operatorname{Li}_3(z),
 \operatorname{Li}_3(1-z)
 \right).
\]
Its coordinate count is independent of $M$.

\section{Archimedean envelope}
For $0<z<1$,
\[
 |\log z|,
 |\log(1-z)|
 \le\log c.
\]
The positive dilogarithm and trilogarithm values on $(0,1)$ are bounded by $\zeta(2)$ and $\zeta(3)$. The three-point identity gives
\[
 \left|\operatorname{Li}_3\left(-\frac ba\right)\right|
 \le
 3\zeta(3)
 +
 \frac{\pi^2}{6}\log c
 +
 \frac23(\log c)^3.
\]
Thus the complex packet has only cubic logarithmic growth. The surrounding motivic and regulator language is informed by the polylogarithmic complexes of Goncharov, the regulator framework discussed by Zagier, and Borel's rank theorem~\cite{Goncharov,ZagierPolylog,Borel}.

This is a genuine nonlinear compression of the displayed complex data. It is not yet an integral compression. A fixed-rank integer lattice carrying a large torsion quotient must pay through covolume, denominator, algebraic degree, or a short vector.

\section{Fixed-rank integral uncertainty}
Let $A\in M_d(\Z)$ be nonsingular and suppose $D\mid\det A$. If every entry satisfies $|A_{ij}|\le B$, then Hadamard's inequality gives
\[
 D\le|\det A|\le d^{d/2}B^d.
\]
Hence
\[
 \boxed{
 B\ge\frac{D^{1/d}}{\sqrt d}.
 }
\]
A small complex period packet does not imply a small integral realization.

\section{Exact finite-place notch}
At an output prime $p^e\parallel c$,
\[
 \beta=1-\frac ca.
\]
Since $p\nmid a$,
\[
 v_p(\beta-1)=e.
\]
For odd $p$,
\[
 \boxed{
 v_p(\log_p\beta)=e.
 }
\]
The corresponding weighted dyadic identity holds on the principal branch when $e\ge2$.

Writing
\[
 \beta=-\prod_{q\mid ab}q^{\nu_q},
\]
we obtain the cross-prime logarithmic filter relation
\[
 \boxed{
 \log_p\beta
 =
 \sum_{q\mid ab}
 \nu_q\log_p\langle q\rangle_p.
 }
\]
The same depth-three path therefore has a small complex envelope and exact deep finite notches.

\section{Weight-three motivation and limitation}
Weight two provides strong functional closure but no nonzero pure rational regulator direction. Weight three is the first plausible rational regulator layer. This motivates the depth-three path packet. It does not prove that weight three suffices uniformly. A complete proof may require adaptive higher weights or a different motivic quotient.

\section{Conditional motivic transfer}
Let $K_\beta$ be the cyclic group generated by the balanced Kummer class. Suppose there exists an integral finite object $\mathcal H_3(a,b,c)$ and a canonical map
\[
 \Theta_3:K_\beta\longrightarrow\mathcal H_3(a,b,c)
\]
satisfying
\[
 |\ker\Theta_3|\le C_\eps R^\eps
\]
and
\[
 |\mathcal H_3(a,b,c)|\le C'_\eps R_aR_bR^\eps.
\]
Then the strong $abc$ conjecture follows after absorbing the logarithmic Kummer kernel.

The missing object must combine finite Kummer depth, the three-point additive relation, an integral motivic or syntomic lattice, and an archimedean period norm.

\chapter{Cross-Prime Logarithmic Coherence}

\section{The signed input exponent signal}
Let
\[
 S_{ab}=\{q:q\mid ab\}.
\]
Define
\[
 \nu_q=v_q(b)-v_q(a).
\]
Then
\[
 -\frac ba=-\prod_{q\mid ab}q^{\nu_q}.
\]
For every output prime $p\mid c$, the same input exponent packet produces the deep principal unit
\[
 1-\frac ca.
\]
The output persistence is therefore a simultaneous cross-prime cancellation phenomenon.

\section{Odd logarithmic rows}
For odd $p^e\parallel c$, define the finite-depth row coefficient
\[
 \ell_{p,e}(q)
 =
 \frac1p\log_p\langle q\rangle_p
 \pmod{p^{e-1}}.
\]
The balanced relation gives
\[
 \boxed{
 \sum_{q\mid ab}
 \nu_q\ell_{p,e}(q)
 \equiv0
 \pmod{p^{e-1}}.
 }
\]
Define the truncated valuation on $\Z/p^{e-1}\Z$ by
\[
 \bar v_p(0)=e-1,
 \qquad
 \bar v_p(x)=\min\{e-1,v_p(\widetilde x)\}
\]
for any integer lift $\widetilde x$ of a nonzero residue. Then
\[
 \boxed{
 \bar v_p(\ell_{p,e}(q))
 =
 \min\{e-1,v_p(q^{p-1}-1)-1\}.
 }
\]
Exceptional entrywise depth is a generalized Wieferich event.

\section{Weighted dyadic row}
For an odd input prime $q$, define
\[
 \ell_{2,e}(q)
 =
 \frac14
 \log_2\left((-1)^{(q-1)/2}q\right)
 \pmod{2^{e-2}}.
\]
With the analogous truncated valuation on $\Z/2^{e-2}\Z$,
\[
 \bar v_2(\ell_{2,e}(q))
 =
 \min\{e-2,v_2(q^2-1)-3\}.
\]
A separate sign packet supplies the missing first dyadic layer. This is the weighted dyadic branch in its final logarithmic form.

\section{Sequential factorization of the powerful part}
The factorization is defined prime by prime. For an odd repeated output prime $p^e\parallel c$, set
\[
 d_p=e-1.
\]
For a repeated dyadic prime $2^e\parallel c$, remove one sign layer and set
\[
 s_2=1,
 \qquad
 d_2=e-2.
\]
If $2\nmid M$, take $s_2=0$. Put $\epsilon_2=2^{s_2}$. Let $K_{\mathrm{Kum}}$ be the kernel size of the balanced Kummer class and put
\[
 r_p=\min\{d_p,v_p(K_{\mathrm{Kum}})\}.
\]
The balanced order formula implies
\[
 K_{\mathrm{Kum}}\mid M/\epsilon_2,
\]
so $K_{\mathrm{Kum}}=\prod_{p\mid c}p^{r_p}$. Write $K=K_{\mathrm{Kum}}$. After dividing the input exponent vector by $K$, let $a_{p,q}$ denote the resulting odd or weighted-dyadic logarithmic row coefficients. Define
\[
 w_p
 =
 \min\left
 \{
 d_p-r_p,
 \min_{q\mid ab}\bar v_p(a_{p,q})
 \right\},
 \qquad
 \chi_p=d_p-r_p-w_p.
\]
The normalized global factors are
\[
 K=\prod_{p\mid c}p^{r_p},
 \qquad
 W=\prod_{p\mid c}p^{w_p},
 \qquad
 C=\prod_{p\mid c}p^{\chi_p}.
\]
The balanced Kummer order formula guarantees that the dyadic sign layer is removed before the principal $2$-adic Kummer depth, so these exponents partition the contact depth. Hence
\[
 \boxed{
 M=\epsilon_2KWC.
 }
\]
The order of removal matters. The local logarithmic defect is measured only after the Kummer preload has been divided out. The resulting row is then divided by its common logarithmic depth, leaving a primitive row modulo $p^{\chi_p}$.

\section{The coherence character}
After normalization, every output prime produces a primitive row
\[
 g_p
 \in
 (\Z/p^{\chi_p}\Z)^{S_{ab}}
\]
satisfying
\[
 g_p\cdot\nu_0
 \equiv0
 \pmod{p^{\chi_p}},
\]
where $\nu_0$ is the Kummer-normalized input exponent vector. The Chinese remainder theorem combines the local rows into one primitive row
\[
 g\in(\Z/C\Z)^{S_{ab}}.
\]
Define the coherence character
\[
 \boxed{
 \chi_{\mathrm{coh}}(x)
 =
 \exp\left(\frac{2\pi i}{C}g\cdot x\right).
 }
\]
Its kernel is the adelic coherence lattice
\[
 \boxed{
 \Lambda_{\mathrm{coh}}
 =
 \{x\in\Z^{S_{ab}}:g\cdot x\equiv0\pmod C\}.
 }
\]
Since $g$ is primitive modulo $C$,
\[
 [\Z^{S_{ab}}:\Lambda_{\mathrm{coh}}]=C
\]
and
\[
 \Z^{S_{ab}}/\Lambda_{\mathrm{coh}}\cong\Z/C\Z.
\]
The normalized exponent vector lies in the lattice.

\section{Worked lattice for \texorpdfstring{$3+125=128$}{3+125=128}}
Here
\[
 M=64
\]
and
\[
 (\epsilon_2,K,W,C)=(2,1,1,32).
\]
The input vector is
\[
 \nu_0=(-1,3)
\]
on the prime coordinates $(3,5)$. The residual row is
\[
 (29,31)\pmod{32}.
\]
Thus
\[
 \boxed{
 \Lambda_{\mathrm{coh}}
 =
 \{(x_3,x_5)\in\Z^2:29x_3+31x_5\equiv0\pmod{32}\}.
 }
\]
Indeed,
\[
 29(-1)+31(3)=64\equiv0\pmod{32}.
\]
The five residual dyadic contact layers are exactly the index of this lattice.

\section{Generalized Wieferich versus collective coherence}
There are two qualitatively different sources of large output depth.

\begin{enumerate}[leftmargin=1.5em]
\item An entrywise defect can make every coefficient in one row divisible by a large power of $p$. This is the factor $W$.
\item Ordinary coefficients can cancel coherently against the specific exponent vector. This is the factor $C$.
\end{enumerate}

Finite data shows that $W$ is usually one, while $C$ is often large. The typical difficult triple is therefore not explained by one exceptional Wieferich edge. It is explained by coordinated cancellation among several ordinary edges.

\section{Why abstract lattice geometry is insufficient}
Let $v\in\Z^s$ be primitive, with $s\ge2$. For every $C\ge2$, there exists a primitive row $g\pmod C$ satisfying
\[
 g\cdot v\equiv0\pmod C.
\]
Thus a fixed short vector can lie in primitive congruence lattices of arbitrarily large index.

\begin{theorem}[Arbitrary-index annihilator]
Abstract lattice geometry cannot bound the coherence index from the length of the known relation vector alone.
\end{theorem}

The proof completes $v$ to a unimodular basis and chooses a primitive row annihilating its first basis coordinate modulo $C$.

The missing theorem must use the arithmetic origin of $g$: its entries are simultaneous $p$-adic logarithms of the actual input primes, and the output primes are exactly the divisors of $a+b$.

\section{Why positive divisibility is circular}
If positive integers $A$ and $B$ satisfy $c\mid A+B$, then
\[
 A+B\ge c,
\]
so
\[
 \max(A,B)\ge c/2.
\]
A direct search for a small positive $S$-unit relation divisible by $c$ cannot produce an $abc$ saving. Signed cancellation, a determinant, a regulator, or a functional equation is necessary.

\section{Exact reformulation of strong \texorpdfstring{$abc$}{abc}}
Since
\[
 c=R_c\epsilon_2KWC
\]
and $\epsilon_2\le2$ while $K\le\log_2c$, the all-$\eps$ strong $abc$ statement is equivalent, after the usual absorption of bounded and logarithmic factors into an arbitrarily small power, to the target
\[
 \boxed{
 WC
 \le
 C_\eps R_aR_bR^\eps.
 }
\]
In the generic finite examples $W=1$, leaving the coherence conductor bound
\[
 C\le C_\eps R_aR_bR^\eps.
\]

\section{Adelic coherence regulator object}
Package the surviving data as
\[
 \boxed{
 \mathfrak R_{a,b,c}
 =
 \left(
 U_{ab},
 \beta^{1/K},
 \{\chi_p\}_{p\mid c},
 \chi_{\mathrm{coh}},
 \Lambda_{\mathrm{coh}},
 \norm{\cdot}_\infty
 \right).
 }
\]
Here $U_{ab}$ is the input $S$-unit group, the local characters are the normalized logarithmic rows, and the archimedean norm records input height.

The open theorem is an adelic slope or regulator inequality bounding the finite conductor $\log C$ by the input radical and an arbitrarily small global leakage term. General simultaneous linear-form theory contains the right kinds of local and archimedean quantities, but known general estimates are much too coarse for the $R^\eps$ scale.

\chapter{The General Prime-Contact Coherence Functor}

\section{Sparse additive relations}
Let
\[
 \epsilon_1n_1+\cdots+\epsilon_rn_r=0,
 \qquad
 \epsilon_i\in\{\pm1\},
\]
be primitive. For every term define
\[
 R_i=\rad(n_i),
 \qquad
 M_i=n_i/R_i.
\]
The prime-contact coherence calculus assigns:

\begin{enumerate}[leftmargin=1.5em]
\item filtered contact packets for every repeated prime power
\item local Witt coordinates and dyadic normalization
\item Kummer classes for balanced multiplicative ratios
\item cross-prime logarithmic filter matrices
\item coherence characters and kernel lattices
\item optional elliptic, hypergeometric, Fermat, polylogarithmic, or automorphic lifts
\end{enumerate}

\section{Canonical operations}
The construction is compatible with several basic operations.

\subsection{Permutation}
Reordering the terms permutes sector labels and leaves the underlying contact and Kummer data unchanged.

\subsection{Sign reversal}
Multiplying the relation by $-1$ reverses the signed exponent packet but leaves the kernel lattice invariant.

\subsection{Primitive reduction}
A common gcd must be removed before forming the contact object. Otherwise one introduces a spurious prime channel shared by every term.

\subsection{Primary decomposition}
Every filtered contact packet and coherence quotient decomposes canonically into its prime-primary factors.

\subsection{Polyphase refinement}
Valuation words may be refined to any modulus. Exact synthesis norms must accompany every refinement.

\subsection{Kummer quotient}
Common powers in a balanced multiplicative relation should be divided out before logarithmic normalization.

\subsection{Functorial gluing}
If two sparse relations share a term, the corresponding contact and Kummer objects should glue across that term. A complete gluing theorem remains open.

\section{Prime-contact data structure}
The reusable object attached to a sparse relation is
\[
 \boxed{
 \mathfrak A(\mathbf n,\boldsymbol\epsilon)
 =
 \left(
 \mathbb W_{\mathbf n},
 \kappa_{\mathbf n},
 \mathcal L_{\mathbf n},
 \Lambda_{\mathrm{coh}},
 \mathcal R_\infty,
 \mathcal M_{\mathrm{glob}}
 \right).
 }
\]
The first four components are finite and local-global arithmetic data. The last two are optional archimedean and global-motivic realizations. Keeping them separate prevents an unproved global comparison from being built into the definition.

\section{Why the calculus extends beyond \texorpdfstring{$abc$}{abc}}
The essential input is not the particular equation $a+b=c$. It is a sparse additive relation with repeated prime depth. This occurs naturally in generalized Fermat equations, perfect-power gaps, recurrence identities, and Bernoulli congruence problems. The later application chapters develop these directions at a programmatic level.
