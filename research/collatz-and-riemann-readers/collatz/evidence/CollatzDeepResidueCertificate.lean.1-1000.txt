import CollatzRefinedResidueCertificate

/-!
# Deep residue refinement for accelerated odd Collatz

This file refines only the nineteen modulo-256 residue families left open in
Proof Atlas v0.13. It introduces thirty-seven parameterized packets at moduli
1024, 2048, and 4096 and proves uniform accelerated descent on every selected
packet. Together with the inherited certificate, this covers exactly 1804 of
the 2048 odd residue classes modulo 4096.

The certificate remains partial: 244 odd classes modulo 4096 remain outside
the selected packet family. No `FiniteDescentCover` or `OddCollatz` theorem is
claimed here.
-/

set_option autoImplicit false

/-- The residue class `287 mod 1024` descends uniformly in 6 accelerated steps. -/
theorem descent_mod_1024_eq_287
    (n : Nat) (hmod : n % 1024 = 287) : it A 6 n < n := by
  obtain ⟨t, rfl⟩ : ∃ t, n = 1024 * t + 287 := by
    refine ⟨n / 1024, ?_⟩
    have hdiv := Nat.mod_add_div n 1024
    omega
  have h1 : A (1024 * t + 287) = 1536 * t + 431 := by
    apply A_of_equation (1024 * t + 287) 1 (1536 * t + 431)
    · omega
    · simp
      omega
  have h2 : A (1536 * t + 431) = 2304 * t + 647 := by
    apply A_of_equation (1536 * t + 431) 1 (2304 * t + 647)
    · omega
    · simp
      omega
  have h3 : A (2304 * t + 647) = 3456 * t + 971 := by
    apply A_of_equation (2304 * t + 647) 1 (3456 * t + 971)
    · omega
    · simp
      omega
  have h4 : A (3456 * t + 971) = 5184 * t + 1457 := by
    apply A_of_equation (3456 * t + 971) 1 (5184 * t + 1457)
    · omega
    · simp
      omega
  have h5 : A (5184 * t + 1457) = 3888 * t + 1093 := by
    apply A_of_equation (5184 * t + 1457) 2 (3888 * t + 1093)
    · omega
    · simp
      omega
  have h6 : A (3888 * t + 1093) = oddPart (729 * t + 205) := by
    unfold A
    have heq : 3 * (3888 * t + 1093) + 1 = 2 ^ 4 * (729 * t + 205) := by
      simp
      omega
    rw [heq, oddPart_pow_mul]
  calc
    it A 6 (1024 * t + 287) = oddPart (729 * t + 205) := by
      simp [it, h1, h2, h3, h4, h5, h6]
    _ ≤ 729 * t + 205 := oddPart_le _
    _ < 1024 * t + 287 := by omega
