import Std

/-!
Exact natural-number Collatz proofs. No axiom or placeholder is used.
`A` removes every factor of two after an odd step. The conjecture is a
separate proposition, not an assumption hidden in any computational rule.
-/
namespace CollatzBlocks

def oddPart (n : Nat) : Nat :=
  if h : 0 < n ∧ n % 2 = 0 then oddPart (n / 2) else n
termination_by n
decreasing_by exact Nat.div_lt_self h.1 (by decide)

theorem oddPart_of_odd (n : Nat) (hn : n % 2 = 1) : oddPart n = n := by
  rw [oddPart]
  split <;> omega

theorem oddPart_two_mul (n : Nat) : oddPart (2 * n) = oddPart n := by
  by_cases hn : n = 0
  · simp [hn]
  · have hp : 0 < 2 * n := by omega
    have he : (2 * n) % 2 = 0 := by omega
    rw [oddPart]
    simp only [hp, he, and_self, dite_true]
    congr 1
    omega

theorem oddPart_pow_mul (a n : Nat) : oddPart (2 ^ a * n) = oddPart n := by
  induction a with
  | zero => simp
  | succ a ih =>
    rw [Nat.pow_succ]
    have h : 2 ^ a * 2 * n = 2 * (2 ^ a * n) := by grind
    rw [h, oddPart_two_mul, ih]

def A (n : Nat) : Nat := oddPart (3 * n + 1)

theorem A_of_equation (n a m : Nat)
    (hm : m % 2 = 1) (heq : 3 * n + 1 = 2 ^ a * m) : A n = m := by
  unfold A
  rw [heq, oddPart_pow_mul, oddPart_of_odd m hm]

def it (f : Nat → Nat) : Nat → Nat → Nat
  | 0, n => n
  | j + 1, n => it f j (f n)

theorem it_add (f : Nat → Nat) (a b n : Nat) :
    it f (a+b) n = it f b (it f a n) := by
  induction a generalizing n with
  | zero => simp [it]
  | succ a ih =>
    rw [show a+1+b = (a+b)+1 by omega]
    simp only [it]
    exact ih (f n)

def ReachesOne (n : Nat) : Prop := ∃ t, it A t n = 1

def OddCollatz : Prop := ∀ n, 0 < n → n % 2 = 1 → ReachesOne n

theorem A_16t11 (t : Nat) : A (16*t+11) = 24*t+17 := by
  apply A_of_equation (16*t+11) 1 (24*t+17)
  · omega
  · simp
    omega

theorem A_24t17 (t : Nat) : A (24*t+17) = 18*t+13 := by
  apply A_of_equation (24*t+17) 2 (18*t+13)
  · omega
  · simp
    omega

theorem pair_16t11 (t : Nat) : it A 2 (16*t+11) = 18*t+13 := by
  simp [it, A_16t11, A_24t17]

theorem pair_16t11_increases (t : Nat) :
    16*t+11 < it A 2 (16*t+11) := by
  rw [pair_16t11]
  omega

/- `2*8^r*q-5` shadows r repetitions of the valuation pair (1,2). -/
theorem repeated_pair (r q : Nat) (hq : 0 < q) :
    it A (2*r) (2*8^r*q-5) = 2*9^r*q-5 := by
  induction r generalizing q with
  | zero => simp [it]
  | succ r ih =>
    have hp : 0 < 8^r*q := Nat.mul_pos (Nat.pow_pos (by decide)) hq
    have hin : 2*8^(r+1)*q-5 = 16*(8^r*q-1)+11 := by
      rw [Nat.pow_succ]
      grind
    have hout : 18*(8^r*q-1)+13 = 2*8^r*(9*q)-5 := by grind
    rw [hin, show 2*(r+1) = 2+2*r by omega, it_add, pair_16t11, hout]
    rw [ih (9*q) (by omega), Nat.pow_succ]
    grind


theorem A_4t3 (t : Nat) : A (4*t+3) = 6*t+5 := by
  apply A_of_equation (4*t+3) 1 (6*t+5)
  · omega
  · simp
    omega

theorem growth_run (j q : Nat) (hq : 0 < q) :
    it A j (2^(j+1)*q-1) = 2*3^j*q-1 := by
  induction j generalizing q with
  | zero => simp [it]
  | succ j ih =>
    have hp : 0 < 2^j*q := Nat.mul_pos (Nat.pow_pos (by decide)) hq
    have hin : 2^(j+1+1)*q-1 = 4*(2^j*q-1)+3 := by
      simp only [Nat.pow_succ]
      grind
    have hout : 6*(2^j*q-1)+5 = 2^(j+1)*(3*q)-1 := by
      rw [Nat.pow_succ]
      grind
    rw [hin]
    simp only [it]
    rw [A_4t3, hout, ih (3*q) (by omega), Nat.pow_succ]
    grind

theorem run_compression (j q b m : Nat) (hq : 0 < q)
    (hm : m % 2 = 1) (he : 3^(j+1)*q = 2^b*m+1) :
    it A (j+1) (2^(j+1)*q-1) = m := by
  rw [it_add A j 1, growth_run j q hq]
  simp only [it]
  apply A_of_equation (2*3^j*q-1) (b+1) m hm
  have hp : 0 < 3^j*q := Nat.mul_pos (Nat.pow_pos (by decide)) hq
  simp only [Nat.pow_succ] at he ⊢
  grind

theorem repeated_pair_prefix (r j q : Nat) (hq : 0 < q) :
    it A (2*j) (2*8^(j+r)*q-5) = 2*9^j*8^r*q-5 := by
  have hp : 0 < 8^r*q := Nat.mul_pos (Nat.pow_pos (by decide)) hq
  have hin : 2*8^(j+r)*q-5 = 2*8^j*(8^r*q)-5 := by
    rw [Nat.pow_add]
    grind
  rw [hin, repeated_pair j (8^r*q) hp]
  grind

theorem repeated_pair_increases (r q : Nat) (hr : 0 < r) (hq : 0 < q) :
    2*8^r*q-5 < it A (2*r) (2*8^r*q-5) := by
  rw [repeated_pair r q hq]
  have hp : 8^r < 9^r := Nat.pow_lt_pow_left (by decide) (by omega)
  have hprod : 2*8^r*q < 2*9^r*q :=
    Nat.mul_lt_mul_of_pos_right (Nat.mul_lt_mul_of_pos_left hp (by decide)) hq
  have h8 : 8 ≤ 8^r := by
    have h := Nat.pow_le_pow_right (n := 8) (by decide) hr
    simpa using h
  have hq' : 8^r ≤ 8^r*q := by
    have h := Nat.mul_le_mul_left (8^r) hq
    simpa using h
  grind

/- The universal pattern includes both growing and contracting accelerated steps. -/
theorem arbitrarily_long_growing_pairs (r : Nat) (hr : 0 < r) :
    ∃ n, 0 < n ∧ n % 2 = 1 ∧ n < it A (2*r) n := by
  refine ⟨2*8^r-5, ?_, ?_, ?_⟩
  · have h8 : 8 ≤ 8^r := by
      have h := Nat.pow_le_pow_right (n := 8) (by decide) hr
      simpa using h
    omega
  · have h8 : 8 ≤ 8^r := by
      have h := Nat.pow_le_pow_right (n := 8) (by decide) hr
      simpa using h
    omega
  · simpa using repeated_pair_increases r 1 hr (by decide)

theorem repeated_pair_exit (r q b m : Nat) (hr : 0 < r) (hq : 0 < q)
    (hm : m % 2 = 1) (he : 3*9^r*q = 2^b*m+7) :
    it A (2*r+1) (2*8^r*q-5) = m := by
  rw [it_add A (2*r) 1, repeated_pair r q hq]
  simp only [it]
  apply A_of_equation (2*9^r*q-5) (b+1) m hm
  have h9 : 9 ≤ 9^r := by
    have h := Nat.pow_le_pow_right (n := 9) (by decide) hr
    simpa using h
  have hq' : 9^r ≤ 9^r*q := by
    have h := Nat.mul_le_mul_left (9^r) hq
    simpa using h
  rw [Nat.pow_succ]
  grind

/- A uniform halving certificate after an arbitrarily long growing pair-run. -/
theorem repeated_pair_exit_halves (r q b m : Nat)
    (hr : 0 < r) (hq : 0 < q) (hb : r+2 ≤ b)
    (he : 3*9^r*q = 2^b*m+7) :
    2*m ≤ (2*8^r*q-5)-1 := by
  have hden : 4*2^r ≤ 2^b := by
    have h := Nat.pow_le_pow_right (n := 2) (by decide) hb
    simp only [Nat.pow_add] at h
    grind
  have hdenm := Nat.mul_le_mul_right m hden
  have hpow : 9^r < 16^r := Nat.pow_lt_pow_left (by decide) (by omega)
  have hy : 9^r*q < (2^r*8^r)*q := by
    have h := Nat.mul_lt_mul_of_pos_right hpow hq
    have heq : 16^r = 2^r*8^r := by
      simpa using Nat.mul_pow 2 8 r
    simpa [heq] using h
  have hcancel : 2^r*(4*m) < 2^r*(3*(8^r*q)) := by grind
  have hsmall : 4*m < 3*(8^r*q) := Nat.lt_of_mul_lt_mul_left hcancel
  have h8 : 8 ≤ 8^r := by
    have h := Nat.pow_le_pow_right (n := 8) (by decide) hr
    simpa using h
  have hq' : 8^r ≤ 8^r*q := by
    have h := Nat.mul_le_mul_left (8^r) hq
    simpa using h
  grind

theorem certified_pair_descent (r q b m : Nat)
    (hr : 0 < r) (hq : 0 < q) (hb : r+2 ≤ b)
    (hm : m % 2 = 1) (he : 3*9^r*q = 2^b*m+7) :
    it A (2*r+1) (2*8^r*q-5) = m ∧ 2*m ≤ (2*8^r*q-5)-1 := by
  exact ⟨repeated_pair_exit r q b m hr hq hm he,
    repeated_pair_exit_halves r q b m hr hq hb he⟩


theorem oddPart_pos_odd (n : Nat) :
    0 < n → 0 < oddPart n ∧ oddPart n % 2 = 1 := by
  refine Nat.strongRecOn n ?_
  intro m ih hm
  rw [oddPart]
  split
  · next h =>
      apply ih (m/2) (Nat.div_lt_self h.1 (by decide))
      omega
  · next h =>
      constructor <;> omega

theorem A_pos_odd (n : Nat) : 0 < A n ∧ A n % 2 = 1 := by
  apply oddPart_pos_odd
  omega

theorem it_A_pos_odd (t n : Nat) (hn : 0 < n) (ho : n % 2 = 1) :
    0 < it A t n ∧ it A t n % 2 = 1 := by
  induction t generalizing n with
  | zero => exact ⟨hn, ho⟩
  | succ t ih =>
      simpa [it] using ih (A n) (A_pos_odd n).1 (A_pos_odd n).2

/- The genuinely unproved premise is displayed explicitly here. -/
theorem collatz_of_universal_descent
    (H : ∀ n, 1 < n → n % 2 = 1 → ∃ t, it A t n < n) : OddCollatz := by
  intro n
  refine Nat.strongRecOn n ?_
  intro m ih hm ho
  by_cases h1 : m = 1
  · subst m
    exact ⟨0, rfl⟩
  · obtain ⟨t, ht⟩ := H m (by omega) ho
    have hpo := it_A_pos_odd t m hm ho
    obtain ⟨u, hu⟩ := ih (it A t m) ht hpo.1 hpo.2
    refine ⟨t+u, ?_⟩
    rw [it_add, hu]

theorem run_exit_halves (j q b m : Nat) (hq : 0 < q) (hb : j+2 ≤ b)
    (he : 3^(j+1)*q = 2^b*m+1) :
    2*m ≤ (2^(j+1)*q-1)-1 := by
  have hden : 2*2^(j+1) ≤ 2^b := by
    have h := Nat.pow_le_pow_right (n := 2) (by decide) hb
    rw [show j+2=(j+1)+1 by omega, Nat.pow_succ] at h
    grind
  have hdenm := Nat.mul_le_mul_right m hden
  have hpow : 3^(j+1) < 4^(j+1) := Nat.pow_lt_pow_left (by decide) (by omega)
  have hy : 3^(j+1)*q < (2^(j+1)*2^(j+1))*q := by
    have h := Nat.mul_lt_mul_of_pos_right hpow hq
    have heq : 4^(j+1) = 2^(j+1)*2^(j+1) := by
      simpa using Nat.mul_pow 2 2 (j+1)
    simpa [heq] using h
  have hc : 2^(j+1)*(2*m) < 2^(j+1)*(2^(j+1)*q) := by grind
  have hs : 2*m < 2^(j+1)*q := Nat.lt_of_mul_lt_mul_left hc
  simp only [Nat.pow_succ] at hs ⊢
  grind

theorem certified_run_descent (j q b m : Nat) (hq : 0 < q) (hb : j+2 ≤ b)
    (hm : m % 2 = 1) (he : 3^(j+1)*q = 2^b*m+1) :
    it A (j+1) (2^(j+1)*q-1) = m ∧ 2*m ≤ (2^(j+1)*q-1)-1 := by
  exact ⟨run_compression j q b m hq hm he, run_exit_halves j q b m hq hb he⟩

/- An unbounded-depth reset into the -5 shadow from the -27 preimage. -/
theorem reset_27_to_5 (q : Nat) (hq : 2 ≤ q) (he : q % 2 = 0) :
    A (16*q-27) = 3*q-5 ∧ A (16*q-27)+5 = 3*q := by
  have ha : A (16*q-27) = 3*q-5 := by
    apply A_of_equation (16*q-27) 4 (3*q-5)
    · omega
    · simp
      omega
  constructor
  · exact ha
  · rw [ha]
    omega


theorem repeated_pair_at (r j q : Nat) (hq : 0 < q) (hj : j ≤ r) :
    it A (2*j) (2*8^r*q-5) = 2*9^j*8^(r-j)*q-5 := by
  have h := repeated_pair_prefix (r-j) j q hq
  rw [show j+(r-j)=r by omega] at h
  exact h

theorem each_completed_pair_grows (r j q : Nat) (hq : 0 < q) (hj : j < r) :
    it A (2*j) (2*8^r*q-5) < it A (2*(j+1)) (2*8^r*q-5) := by
  have hp : 0 < 9^j*8^(r-j-1)*q :=
    Nat.mul_pos (Nat.mul_pos (Nat.pow_pos (by decide)) (Nat.pow_pos (by decide))) hq
  have hx : it A (2*j) (2*8^r*q-5) = 16*(9^j*8^(r-j-1)*q-1)+11 := by
    rw [repeated_pair_at r j q hq (by omega)]
    rw [show r-j = (r-j-1)+1 by omega, Nat.pow_succ]
    grind
  rw [show 2*(j+1)=2*j+2 by omega, it_add, hx]
  exact pair_16t11_increases (9^j*8^(r-j-1)*q-1)

theorem arbitrarily_many_completed_blocks_grow (r : Nat) (hr : 0 < r) :
    ∃ n, 0 < n ∧ n % 2 = 1 ∧
      ∀ j, j < r → it A (2*j) n < it A (2*(j+1)) n := by
  refine ⟨2*8^r-5, ?_, ?_, ?_⟩
  · have h8 : 8 ≤ 8^r := by
      have h := Nat.pow_le_pow_right (n := 8) (by decide) hr
      simpa using h
    omega
  · have h8 : 8 ≤ 8^r := by
      have h := Nat.pow_le_pow_right (n := 8) (by decide) hr
      simpa using h
    omega
  · intro j hj
    simpa using each_completed_pair_grows r j 1 (by decide) hj

/- A concrete infinite progression, not merely a finite test vector. -/
theorem progression_256t203 (t : Nat) :
    it A 3 (256*t+203) = 54*t+43 ∧ 2*(54*t+43) ≤ (256*t+203)-1 := by
  have h := certified_pair_descent 1 (16*t+13) 3 (54*t+43)
    (by decide) (by omega) (by decide) (by omega) (by simp; omega)
  simp at h
  grind

theorem no_universal_two_step_descent :
    ¬ (∀ n, 1 < n → n % 2 = 1 → it A 2 n < n) := by
  intro h
  have hh := h 11 (by decide) (by decide)
  have he := pair_16t11 0
  simp at he
  omega


/- A growing (1,2)-pair shadow followed by a reset into an arbitrarily deep
   a=1 run. Both r and h are unbounded. No convergence premise is used. -/
theorem pair_reset_compression (r h q v b m : Nat)
    (hr : 0 < r) (hh : 0 < h) (hq : 0 < q) (hv : 0 < v)
    (hm : m % 2 = 1)
    (hreset : 9^r*q = 2^h*v+1)
    (hexit : 3^(h+2)*v = 2^b*m+1) :
    it A (2*r+(h+2)) (4*8^r*q-5) = m := by
  have hin : 4*8^r*q-5 = 2*8^r*(2*q)-5 := by grind
  rw [hin, it_add, repeated_pair r (2*q) (by omega)]
  have hmid : 2*9^r*(2*q)-5 = 2^(h+2)*v-1 := by
    simp only [Nat.pow_add]
    grind
  rw [hmid]
  exact run_compression (h+1) v b m hv hm hexit

theorem pair_reset_exit_halves (r h q v b m : Nat)
    (hr : 0 < r) (hq : 0 < q) (hv : 0 < v)
    (hb : r+h+4 ≤ b)
    (hreset : 9^r*q = 2^h*v+1)
    (hexit : 3^(h+2)*v = 2^b*m+1) :
    2*m ≤ (4*8^r*q-5)-1 := by
  have hden : 2^(h+2)*2^(r+2) ≤ 2^b := by
    have hh : (h+2)+(r+2) ≤ b := by omega
    have ht := Nat.pow_le_pow_right (n := 2) (by decide) hh
    simpa [Nat.pow_add] using ht
  have hdenm := Nat.mul_le_mul_right m hden
  have hpow : 3^(h+2) < 4^(h+2) := Nat.pow_lt_pow_left (by decide) (by omega)
  have hy : 3^(h+2)*v < (2^(h+2)*2^(h+2))*v := by
    have ht := Nat.mul_lt_mul_of_pos_right hpow hv
    have heq : 4^(h+2) = 2^(h+2)*2^(h+2) := by
      simpa using Nat.mul_pow 2 2 (h+2)
    simpa [heq] using ht
  have hc : 2^(h+2)*(2^(r+2)*m) < 2^(h+2)*(2^(h+2)*v) := by grind
  have hs : 2^(r+2)*m < 2^(h+2)*v := Nat.lt_of_mul_lt_mul_left hc
  have hmid : 2^r*m < 9^r*q := by
    simp only [Nat.pow_add] at hs
    grind
  have h916 : 9^r*q < (2^r*8^r)*q := by
    have ht := Nat.mul_lt_mul_of_pos_right
      (Nat.pow_lt_pow_left (a:=9) (b:=16) (n:=r) (by decide) (by omega)) hq
    have heq : 16^r = 2^r*8^r := by simpa using Nat.mul_pow 2 8 r
    simpa [heq] using ht
  have hc2 : 2^r*m < 2^r*(8^r*q) := by grind
  have hmX : m < 8^r*q := Nat.lt_of_mul_lt_mul_left hc2
  have h8 : 8 ≤ 8^r := by
    have ht := Nat.pow_le_pow_right (n := 8) (by decide) hr
    simpa using ht
  have hq' : 8^r ≤ 8^r*q := by
    have ht := Nat.mul_le_mul_left (8^r) hq
    simpa using ht
  grind

theorem certified_pair_reset_descent (r h q v b m : Nat)
    (hr : 0 < r) (hh : 0 < h) (hq : 0 < q) (hv : 0 < v)
    (hb : r+h+4 ≤ b) (hm : m % 2 = 1)
    (hreset : 9^r*q = 2^h*v+1)
    (hexit : 3^(h+2)*v = 2^b*m+1) :
    it A (2*r+(h+2)) (4*8^r*q-5) = m ∧ 2*m ≤ (4*8^r*q-5)-1 := by
  exact ⟨pair_reset_compression r h q v b m hr hh hq hv hm hreset hexit,
    pair_reset_exit_halves r h q v b m hr hq hv hb hreset hexit⟩
