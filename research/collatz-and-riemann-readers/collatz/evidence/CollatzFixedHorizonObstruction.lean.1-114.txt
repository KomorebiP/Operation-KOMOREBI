import CollatzFiniteCoverBridge
import Mathlib.Data.Finset.Lattice.Fold
import Mathlib.Tactic.Ring

/-! Exact all-ones valuation prefixes obstruct every finite fixed-horizon
    descent cover. This does not obstruct input-dependent unbounded times. -/
set_option autoImplicit false
namespace OperationKomorebi.CollatzFixedHorizonObstruction
open CollatzBlocks CollatzFiniteCoverBridge

/-- A prefix of length j inside a longer all-ones valuation word. -/
theorem growth_prefix (j r q : Nat) (hq : 0 < q) :
    it A j (2 ^ (j + r + 1) * q - 1) =
      2 * 3 ^ j * 2 ^ r * q - 1 := by
  have he : 2 ^ (j + r + 1) * q = 2 ^ (j + 1) * (2 ^ r * q) := by
    rw [show j + r + 1 = (j + 1) + r by omega, Nat.pow_add]
    ring
  rw [he, growth_run j (2 ^ r * q) (Nat.mul_pos (Nat.pow_pos (by decide)) hq)]
  congr 1
  ring

/-- Every such prefix is at least its own source, including prefix zero. -/
theorem growth_prefix_nondecreasing (j r q : Nat) (hq : 0 < q) :
    2 ^ (j + r + 1) * q - 1 ≤ it A j (2 ^ (j + r + 1) * q - 1) := by
  rw [growth_prefix j r q hq]
  have hp : 2 ^ j ≤ 3 ^ j := Nat.pow_le_pow_left (by decide) j
  have hmul := Nat.mul_le_mul_right (2 ^ r * q) (Nat.mul_le_mul_left 2 hp)
  have he : 2 ^ (j + r + 1) * q = 2 * 2 ^ j * 2 ^ r * q := by
    rw [show j + r + 1 = (j + 1) + r by omega, Nat.pow_add, Nat.pow_succ]
    ring
  rw [he]
  apply Nat.sub_le_sub_right
  simpa only [Nat.mul_assoc] using hmul

/-- One explicit positive odd witness escapes every descent time up to K. -/
theorem witness_no_descent_up_to (K j : Nat) (hj : j ≤ K) :
    2 ^ (K + 2) - 1 ≤ it A j (2 ^ (K + 2) - 1) := by
  have he : j + (K + 1 - j) + 1 = K + 2 := by omega
  simpa only [he, Nat.mul_one] using
    growth_prefix_nondecreasing j (K + 1 - j) 1 (by decide)

/-- There is no finite universal strict-descent horizon. -/
theorem no_uniform_descent_horizon (K : Nat) :
    ∃ n : Nat, 1 < n ∧ n % 2 = 1 ∧ ∀ j ≤ K, n ≤ it A j n := by
  refine ⟨2 ^ (K + 2) - 1, ?_, ?_, fun j hj => witness_no_descent_up_to K j hj⟩
  · have hp : 0 < 2 ^ K := Nat.pow_pos (by decide)
    simp only [Nat.pow_add] at *
    omega
  · have hp : 0 < 2 ^ K := Nat.pow_pos (by decide)
    simp only [Nat.pow_add] at *
    omega

/-- The v0.11 finite-cover antecedent is uninhabited: every rule has a fixed
    step count, and finitely many such counts have a common upper bound. -/
theorem no_finite_fixed_depth_cover {ι : Type} [Fintype ι] :
    IsEmpty (FiniteDescentCover ι) := by
  classical
  refine ⟨fun C => ?_⟩
  let K := Finset.univ.sup (fun i => (C.rule i).steps)
  obtain ⟨n, hn, ho, hslow⟩ := no_uniform_descent_horizon K
  obtain ⟨i, hi⟩ := C.covers n hn ho
  have hbound : (C.rule i).steps ≤ K := Finset.le_sup (f := fun a : ι => (C.rule a).steps) (Finset.mem_univ i)
  exact (Nat.not_lt_of_ge (hslow _ hbound)) ((C.rule i).descent n hi hn ho)

/-- A finite table assigning a fixed depth to each residue also cannot cover
    all odd inputs, for any modulus. -/
theorem no_residue_fixed_depth_cover (q : Nat) :
    IsEmpty (ResidueDescentCover q) := by
  classical
  refine ⟨fun C => ?_⟩
  let K := Finset.univ.sup C.steps
  obtain ⟨n, hn, ho, hslow⟩ := no_uniform_descent_horizon K
  let i : Fin q := ⟨n % q, Nat.mod_lt n C.q_pos⟩
  have hbound : C.steps i ≤ K := Finset.le_sup (Finset.mem_univ i)
  exact (Nat.not_lt_of_ge (hslow _ hbound)) (C.descent n hn ho)

/-- Arbitrarily long delays persist inside a single listed v0.14 residual
    class: 4095 modulo 4096. -/
theorem residual_4095_arbitrary_delay (K : Nat) :
    ∃ n : Nat, 1 < n ∧ n % 4096 = 4095 ∧ n % 2 = 1 ∧
      ∀ j ≤ K, n ≤ it A j n := by
  refine ⟨2 ^ (K + 12) - 1, ?_, ?_, ?_, ?_⟩
  · have hp : 0 < 2 ^ K := Nat.pow_pos (by decide)
    simp only [Nat.pow_add]
    omega
  · have hp : 0 < 2 ^ K := Nat.pow_pos (by decide)
    simp only [Nat.pow_add]
    omega
  · have hp : 0 < 2 ^ K := Nat.pow_pos (by decide)
    simp only [Nat.pow_add]
    omega
  · intro j hj
    exact witness_no_descent_up_to (K + 10) j (by omega)

/-- Any correct input-dependent descent selector must have unbounded times. -/
theorem valid_descent_selector_unbounded (T : Nat → Nat)
    (hT : ∀ n, 1 < n → n % 2 = 1 → it A (T n) n < n) (K : Nat) :
    ∃ n, 1 < n ∧ n % 2 = 1 ∧ K < T n := by
  obtain ⟨n, hn, ho, hs⟩ := no_uniform_descent_horizon K
  refine ⟨n, hn, ho, ?_⟩
  by_contra h
  exact (Nat.not_lt_of_ge (hs (T n) (by omega))) (hT n hn ho)

/-- Even a correct selector restricted to the single 4095 packet cannot have
    a finite upper bound. -/
theorem residual_descent_selector_unbounded (T : Nat → Nat)
    (hT : ∀ n, 1 < n → n % 4096 = 4095 → it A (T n) n < n) (K : Nat) :
    ∃ n, 1 < n ∧ n % 4096 = 4095 ∧ K < T n := by
  obtain ⟨n, hn, hr, _, hs⟩ := residual_4095_arbitrary_delay K
  refine ⟨n, hn, hr, ?_⟩
  by_contra h
  exact (Nat.not_lt_of_ge (hs (T n) (by omega))) (hT n hn hr)

end OperationKomorebi.CollatzFixedHorizonObstruction
