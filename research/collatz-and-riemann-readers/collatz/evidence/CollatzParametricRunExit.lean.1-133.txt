import CollatzUnboundedRunFamily

/-! Uniform lifts of compressed Collatz exits to arithmetic progressions.
These are input-dependent certificates, not a global cover. -/
set_option autoImplicit false
namespace OperationKomorebi.CollatzParametricRunExit
open CollatzBlocks CollatzFixedHorizonObstruction CollatzUnboundedRunFamily

def runStart (j q : Nat) : Nat := 2^(j+1)*q-1
def liftedQuotient (q b s : Nat) : Nat := q+2^(b+1)*s
def liftedTarget (j m s : Nat) : Nat := m+2*3^(j+1)*s

theorem lifted_quotient_pos (q b s : Nat) (hq : 0 < q) :
    0 < liftedQuotient q b s := by unfold liftedQuotient; omega

theorem lifted_target_odd (j m s : Nat) (hm : m % 2 = 1) :
    liftedTarget j m s % 2 = 1 := by
  simp [liftedTarget, Nat.add_mod, Nat.mul_mod, hm]

theorem lifted_exit_equation (j q b m s : Nat)
    (he : 3^(j+1)*q = 2^b*m+1) :
    3^(j+1)*liftedQuotient q b s = 2^b*liftedTarget j m s+1 := by
  unfold liftedQuotient liftedTarget
  rw [Nat.mul_add, he, Nat.pow_succ]
  ring

theorem source_progression (j q b s : Nat) (hq : 0 < q) :
    runStart j (liftedQuotient q b s) = runStart j q+2^(j+b+2)*s := by
  have hp : 0 < 2^(j+1)*q := Nat.mul_pos (by positivity) hq
  have he : 2^(j+1)*2^(b+1) = 2^(j+b+2) := by
    rw [← Nat.pow_add]
    congr 1
    omega
  unfold runStart liftedQuotient
  rw [Nat.mul_add, ← Nat.mul_assoc, he]
  omega

/-- Every natural progression parameter has a certified exact run exit. -/
theorem lifted_run_exit (j q b m s : Nat) (hq : 0 < q)
    (hm : m % 2 = 1) (he : 3^(j+1)*q = 2^b*m+1) :
    it A (j+1) (runStart j q+2^(j+b+2)*s) = liftedTarget j m s := by
  rw [← source_progression j q b s hq]
  exact run_compression j _ b _ (lifted_quotient_pos q b s hq)
    (lifted_target_odd j m s hm) (lifted_exit_equation j q b m s he)

/-- A sufficient slope condition makes the base descent uniform over the
entire progression. It is a hypothesis, not asserted for arbitrary runs. -/
theorem lifted_target_lt_source (j q b m s : Nat)
    (hbase : m < runStart j q) (hslope : 3^(j+1) ≤ 2^(j+b+1)) :
    liftedTarget j m s < runStart j q+2^(j+b+2)*s := by
  have hc : 2*3^(j+1) ≤ 2^(j+b+2) := by
    have h := Nat.mul_le_mul_left 2 hslope
    have he : j+b+2 = (j+b+1)+1 := by omega
    rw [he, Nat.pow_succ]
    omega
  have h := Nat.mul_le_mul_right s hc
  unfold liftedTarget
  omega

/-- The first strict descent, not just some descent, occurs at j+1. -/
theorem progression_exact_first_descent (j q b m s : Nat) (hq : 0 < q)
    (hm : m % 2 = 1) (he : 3^(j+1)*q = 2^b*m+1)
    (hbase : m < runStart j q) (hslope : 3^(j+1) ≤ 2^(j+b+1)) :
    (∀ t ≤ j, runStart j q+2^(j+b+2)*s ≤
      it A t (runStart j q+2^(j+b+2)*s)) ∧
    it A (j+1) (runStart j q+2^(j+b+2)*s) = liftedTarget j m s ∧
    liftedTarget j m s < runStart j q+2^(j+b+2)*s := by
  refine ⟨?_, lifted_run_exit j q b m s hq hm he,
    lifted_target_lt_source j q b m s hbase hslope⟩
  intro t ht
  rw [← source_progression j q b s hq]
  have hpow : t+(j-t)+1 = j+1 := by omega
  simpa only [hpow, runStart] using
    growth_prefix_nondecreasing t (j-t) (liftedQuotient q b s)
      (lifted_quotient_pos q b s hq)

/-- Termination transfers from the computed exit; it is not assumed for
arbitrary exits. The number of added steps is explicitly j+1. -/
theorem termination_of_lifted_target (j q b m s t : Nat) (hq : 0 < q)
    (hm : m % 2 = 1) (he : 3^(j+1)*q = 2^b*m+1)
    (ht : it A t (liftedTarget j m s) = 1) :
    it A ((j+1)+t) (runStart j q+2^(j+b+2)*s) = 1 := by
  rw [it_add, lifted_run_exit j q b m s hq hm he, ht]

theorem parameter_le_three_power (j : Nat) : j+1 ≤ 3^j := by
  induction j with
  | zero => norm_num
  | succ j ih => rw [Nat.pow_succ]; omega

theorem seed_slope_bound (j : Nat) : 3^(j+1) ≤ 2^(j+3^j+1) := by
  calc
    3^(j+1) ≤ 4^(j+1) := Nat.pow_le_pow_left (by decide : 3 ≤ 4) (j+1)
    _ = 2^((j+1)+(j+1)) := by
      calc
        4^(j+1) = (2*2)^(j+1) := rfl
        _ = 2^(j+1)*2^(j+1) := Nat.mul_pow 2 2 (j+1)
        _ = 2^((j+1)+(j+1)) := (Nat.pow_add 2 (j+1) (j+1)).symm
    _ ≤ 2^(j+3^j+1) := Nat.pow_le_pow_right (by decide)
      (by have h := parameter_le_three_power j; omega)

/-- A concrete two-parameter family: each former terminating seed is the
start of an infinite progression with the same exact first-descent time. -/
theorem seed_progression_exact_first_descent (j s : Nat) (hj : 0 < j) :
    (∀ t ≤ j, start j+2^(j+3^j+2)*s ≤
      it A t (start j+2^(j+3^j+2)*s)) ∧
    it A (j+1) (start j+2^(j+3^j+2)*s) = 1+2*3^(j+1)*s ∧
    1+2*3^(j+1)*s < start j+2^(j+3^j+2)*s := by
  exact progression_exact_first_descent j (runQuotient j) (3^j) 1 s
    (quotient_pos j) (by decide) (by simpa using quotient_equation j)
    (start_admissible j hj).1 (seed_slope_bound j)

theorem seed_progression_admissible (j s : Nat) (hj : 0 < j) :
    1 < start j+2^(j+3^j+2)*s ∧ (start j+2^(j+3^j+2)*s) % 2 = 1 := by
  have ha := start_admissible j hj
  constructor
  · omega
  · simp [Nat.add_mod, Nat.mul_mod, Nat.pow_mod, ha.2]

theorem seed_progression_injective (j : Nat) :
    Function.Injective (fun s : Nat => start j+2^(j+3^j+2)*s) := by
  intro s t h
  have hp : 0 < 2^(j+3^j+2) := by positivity
  exact Nat.eq_of_mul_eq_mul_left hp (Nat.add_left_cancel h)

theorem seed_progression_in_residual (j s : Nat) (hj : 11 ≤ j) :
    (start j+2^(j+3^j+2)*s) % 4096 = 4095 := by
  have hp : 0 < 3^j := by positivity
  have h : 12 ≤ j+3^j+2 := by omega
  have he : j+3^j+2 = 12+(j+3^j+2-12) := by omega
  rw [Nat.add_mod, family_in_residual_4095 j hj, he, Nat.pow_add]
  norm_num [Nat.mul_mod]

end OperationKomorebi.CollatzParametricRunExit
