def twoModNine : (ZMod 9)ˣ :=
  ⟨2,5,by decide,by decide⟩

theorem teichmuller_two_mod_nine :
    (FiniteUnits.omega 3 1 twoModNine : ZMod 9) = 8 := by decide

theorem principal_two_mod_nine :
    (FiniteUnits.principal 3 1 twoModNine : ZMod 9) = 7 := by decide

theorem ordinary_residue_is_not_lifted_root : (2 : ZMod 9)^2 ≠ 1 := by decide

def threeModEight : (ZMod 8)ˣ :=
  ⟨3,3,by decide,by decide⟩

theorem dyadic_three_sign :
    (DyadicUnits.sign 1 threeModEight : ZMod 8) = 7 := by decide

theorem dyadic_three_principal :
    (DyadicUnits.principal 1 threeModEight : ZMod 8) = 5 := by decide

theorem dyadic_three_reconstruction :
    (7 : ZMod 8)*5 = 3 ∧ (5 : ZMod 4)=1 := by decide

/-- Coprimality is necessary for the coordinate map to be injective. -/
theorem noncoprime_components_do_not_separate :
    Nat.ModEq 2 0 2 ∧ Nat.ModEq 2 0 2 ∧ ¬ Nat.ModEq (2*2) 0 2 := by decide

theorem composite_fermat_failure : (2 : ZMod 9)^8 ≠ 1 := by decide

/-- The p=3,m=2 residual row is derived from the constructed forward substitution. -/
theorem reduced_first_row : WindowIndependence.reduced 3 2 1 = (2 : ZMod 3) := by
  have h := WindowIndependence.reduced_row 3 2 1
  norm_num [Finset.sum_range_succ] at h
  have he : WindowIndependence.reduced 3 2 1 = -(1 : ZMod 3) := by linear_combination h
  exact he.trans (by decide)

theorem reduced_second_row : WindowIndependence.reduced 3 2 2 = (2 : ZMod 3) := by
  have h := WindowIndependence.reduced_row 3 2 2
  norm_num [Finset.sum_range_succ, reduced_first_row] at h
  have he : WindowIndependence.reduced 3 2 2 = -(13 : ZMod 3) := by linear_combination h
  exact he.trans (by decide)

/-- All three standard-window addresses have this same nonzero first residue. -/
theorem standard_window_example (j : ℕ) (hj : j < 3) :
    PadicInt.toZMod (WindowIndependence.coeffPadic 3 2 j hj 1) = (2 : ZMod 3) := by
  rw [WindowIndependence.residue_independence, reduced_first_row]

/-- The forbidden window address has an actual denominator divisible by p. -/
theorem outside_window_denominator :
    3 ∣ (Congruential.coeff 6 3 1).den := by
  rw [Congruential.coefficient_recurrence 6 3 (by decide : 0 < 1)]
  norm_num [Finset.sum_range_succ, Nat.choose]

/-- The window restriction is necessary for the original rational coefficients themselves. -/
theorem outside_window_not_integral :
    ¬ ‖(Congruential.coeff 6 3 1 : ℚ_[3])‖ ≤ 1 := by
  intro h
  have hu := PadicInt.isUnit_den (p := 3) (Congruential.coeff 6 3 1) h
  have hc := PadicInt.norm_natCast_eq_one_iff.mp (PadicInt.isUnit_iff.mp hu)
  exact (Nat.prime_three.coprime_iff_not_dvd.mp hc) outside_window_denominator

end KomorebiPaperIX.FiniteArithmeticExamples
