module

public import CompanionPackets
public import Mathlib.Data.Nat.Fib.Basic
public import Mathlib.Tactic.NormNum.NatFib

@[expose] public section
noncomputable section
set_option autoImplicit false

namespace KomorebiPaperIX.FibonacciPackets

open Matrix Polynomial

/-- The integer Fibonacci companion matrix in the state order (F_(n+1), F_n). -/
def companion : Matrix (Fin 2) (Fin 2) ℤ := !![1, 1; 1, 0]

/-- The standard Lucas sequence, independently defined by its initial values and recurrence. -/
def lucas : ℕ → ℤ
  | 0 => 2
  | 1 => 1
  | n+2 => lucas n + lucas (n+1)

/-- Actual Mathlib Fibonacci values, cast into Z to preserve signed recurrence arithmetic. -/
def state (n : ℕ) : Fin 2 → ℤ := ![(Nat.fib (n+1):ℤ), (Nat.fib n:ℤ)]

def initial : Fin 2 → ℤ := ![1,0]

/-- The independently defined Fibonacci values satisfy the companion dynamics. -/
theorem state_step (n : ℕ) : state (n+1) = companion *ᵥ state n := by
  ext i
  fin_cases i <;> simp [state, companion, Matrix.mulVec, dotProduct, Fin.sum_univ_two,
    Nat.fib_add_two, add_comm]

@[simp] theorem state_zero : state 0 = initial := by
  simp [state, initial]

theorem state_is_orbit (n : ℕ) : state n = Companion.orbit companion initial n := by
  simpa only [state_zero] using Companion.orbit_of_recurrence companion state state_step n

/-- The packet state really contains F_(Nk+j), not a newly postulated sequence. -/
theorem packet_is_fibonacci (N j k : ℕ) :
    Companion.packet companion initial N j k 1 = (Nat.fib (N*k+j):ℤ) := by
  rw [Companion.packet, ← state_is_orbit]
  rfl

/-- Matrix identity from which the Lucas trace recurrence follows. -/
theorem companion_square : companion^2 = companion + 1 := by
  ext i j
  fin_cases i <;> fin_cases j <;> norm_num [companion, pow_two, Matrix.mul_apply, Fin.sum_univ_two]

theorem companion_power_add_two (n : ℕ) :
    companion^(n+2) = companion^n + companion^(n+1) := by
  rw [pow_add, companion_square, mul_add, mul_one, ← pow_succ]
  exact add_comm _ _

/-- No Binet roots are needed: the actual trace agrees with the Lucas recurrence. -/
theorem trace_power_is_lucas (n : ℕ) : (companion^n).trace = lucas n := by
  induction n using Nat.twoStepInduction with
  | zero => norm_num [lucas, Matrix.trace, Fin.sum_univ_two]
  | one => norm_num [lucas, companion, Matrix.trace, Fin.sum_univ_two]
  | more n ih0 ih1 => rw [companion_power_add_two, Matrix.trace_add, ih0, ih1, lucas]

theorem det_companion : companion.det = -1 := by norm_num [companion, Matrix.det_fin_two]

theorem det_power (N : ℕ) : (companion^N).det = (-1:ℤ)^N := by
  rw [Matrix.det_pow, det_companion]

/-- The actual characteristic polynomial of the jump, with its correct signed constant term. -/
theorem jump_charpoly (N : ℕ) :
    (companion^N).charpoly = X^2 - C (lucas N)*X + C ((-1:ℤ)^N) := by
  rw [Matrix.charpoly_fin_two, trace_power_is_lucas, det_power]

/-- Corollary 9.2: every arithmetic Fibonacci subsequence, all steps and offsets. -/
theorem fibonacci_packet_recurrence (N j k : ℕ) :
    (Nat.fib (N*(k+2)+j):ℤ) =
      lucas N * (Nat.fib (N*(k+1)+j):ℤ) - (-1:ℤ)^N * (Nat.fib (N*k+j):ℤ) := by
  have h := Companion.packet_coordinate_recurrence companion initial N j k 1
  simp only [Fintype.card_fin, jump_charpoly, Finset.sum_range_succ,
    Finset.sum_range_zero, zero_add, add_zero, packet_is_fibonacci] at h
  norm_num at h
  linear_combination h

/-- The full state vector satisfies the same recurrence, not only one coordinate. -/
theorem fibonacci_state_packet_recurrence (N j k : ℕ) :
    state (N*(k+2)+j) =
      lucas N • state (N*(k+1)+j) - ((-1:ℤ)^N) • state (N*k+j) := by
  have h := Companion.packet_charpoly_recurrence companion initial N j k
  simp only [Fintype.card_fin, jump_charpoly, Finset.sum_range_succ,
    Finset.sum_range_zero, zero_add, add_zero, Companion.packet, ← state_is_orbit] at h
  norm_num at h
  simpa [sub_eq_add_neg, add_comm, neg_add_rev] using h

/-- The first two sampled values determine the entire packet for a fixed jump. -/
theorem fibonacci_packet_unique (N j : ℕ) (u : ℕ → ℤ)
    (h0 : u 0 = Nat.fib j) (h1 : u 1 = Nat.fib (N+j))
    (hu : ∀ k, u (k+2) = lucas N * u (k+1) - (-1:ℤ)^N * u k) :
    ∀ k, u k = Nat.fib (N*k+j) := by
  intro k
  induction k using Nat.twoStepInduction with
  | zero => simpa using h0
  | one => simpa using h1
  | more k ih0 ih1 => rw [hu, ih0, ih1, fibonacci_packet_recurrence]

/-- An even jump carries a minus sign on the previous state. -/
theorem fibonacci_even_step (N j k : ℕ) (hN : Even N) :
    (Nat.fib (N*(k+2)+j):ℤ) =
      lucas N * (Nat.fib (N*(k+1)+j):ℤ) - (Nat.fib (N*k+j):ℤ) := by
  rw [fibonacci_packet_recurrence, hN.neg_one_pow, one_mul]

/-- An odd jump carries a plus sign on the previous state. -/
theorem fibonacci_odd_step (N j k : ℕ) (hN : Odd N) :
    (Nat.fib (N*(k+2)+j):ℤ) =
      lucas N * (Nat.fib (N*(k+1)+j):ℤ) + (Nat.fib (N*k+j):ℤ) := by
  rw [fibonacci_packet_recurrence, hN.neg_one_pow]
  ring

/-- Lucas numbers also occur as a Fibonacci linear combination, including N=0. -/
theorem lucas_is_fibonacci_combination (n : ℕ) :
    lucas n = 2*(Nat.fib (n+1):ℤ) - (Nat.fib n:ℤ) := by
  induction n using Nat.twoStepInduction with
  | zero => norm_num [lucas]
  | one => norm_num [lucas]
  | more n ih0 ih1 =>
    rw [lucas, ih0, ih1]
    push_cast
    simp only [show n+2+1=(n+1)+2 by omega, Nat.fib_add_two, Nat.cast_add]
    ring

end KomorebiPaperIX.FibonacciPackets
