module

public import ResiduePackets
public import Mathlib.RingTheory.PowerSeries.Inverse
public import Mathlib.Data.Nat.Choose.Cast
public import Mathlib.Algebra.BigOperators.NatAntidiagonal

@[expose] public section
noncomputable section
set_option autoImplicit false

namespace KomorebiPaperIX.EGF

open PowerSeries

def encode (a : ℕ → ℚ) : PowerSeries ℚ :=
  PowerSeries.mk (fun n => a n / (n.factorial : ℚ))

def decode (A : PowerSeries ℚ) : ℕ → ℚ :=
  fun n => PowerSeries.coeff n A * (n.factorial : ℚ)

@[simp] theorem encode_coeff (a : ℕ → ℚ) (n : ℕ) :
    coeff n (encode a) = a n / (n.factorial : ℚ) := by simp [encode]

@[simp] theorem decode_encode (a : ℕ → ℚ) : decode (encode a) = a := by
  funext n
  simp [decode, Nat.factorial_ne_zero]

@[simp] theorem encode_decode (A : PowerSeries ℚ) : encode (decode A) = A := by
  ext n
  simp [decode, Nat.factorial_ne_zero]

theorem encode_add (a b : ℕ → ℚ) : encode (a+b) = encode a + encode b := by
  ext n
  simp [add_div]

theorem decode_add (A B : PowerSeries ℚ) : decode (A+B) = decode A + decode B := by
  funext n
  simp [decode, add_mul]

@[simp] theorem encode_zero : encode 0 = 0 := by ext n; simp

@[simp] theorem decode_zero : decode 0 = 0 := by funext n; simp [decode]

theorem egf_product (h a : ℕ → ℚ) (n : ℕ) :
    decode (encode h * encode a) n =
      ∑ r ∈ Finset.range (n+1), (n.choose r : ℚ) * h r * a (n-r) := by
  simp only [decode, coeff_mul, Finset.Nat.sum_antidiagonal_eq_sum_range_succ_mk,
    encode_coeff, Finset.sum_mul]
  apply Finset.sum_congr rfl
  intro r hr
  rw [Nat.cast_choose ℚ (Finset.mem_range_succ_iff.mp hr)]
  ring

def transfer (H : PowerSeries ℚ) (a : ℕ → ℚ) : ℕ → ℚ := decode (H * encode a)

theorem transfer_comp (H K : PowerSeries ℚ) (a : ℕ → ℚ) :
    transfer H (transfer K a) = transfer (H*K) a := by
  simp [transfer, mul_assoc]

@[simp] theorem transfer_one (a : ℕ → ℚ) : transfer 1 a = a := by simp [transfer]

theorem transfer_add (H : PowerSeries ℚ) (a b : ℕ → ℚ) :
    transfer H (a+b) = transfer H a + transfer H b := by
  simp [transfer, encode_add, mul_add, decode_add]

@[simp] theorem transfer_zero (H : PowerSeries ℚ) : transfer H 0 = 0 := by simp [transfer]

theorem transfer_sum {ι : Type*} (H : PowerSeries ℚ) (s : Finset ι) (a : ι → ℕ → ℚ) :
    transfer H (∑ i ∈ s, a i) = ∑ i ∈ s, transfer H (a i) := by
  classical
  induction s using Finset.induction_on with
  | empty => simp
  | @insert i s hi ih => simp [Finset.sum_insert hi, transfer_add, ih]

/-- The inverse is constructed by the power-series inverse, not a supplied inverse law. -/
def coordinateEquiv (h : ℕ → ℚ) (h0 : h 0 = 1) : (ℕ → ℚ) ≃ (ℕ → ℚ) where
  toFun := transfer (encode h)
  invFun := transfer (encode h)⁻¹
  left_inv a := by
    rw [transfer_comp, PowerSeries.inv_mul_cancel, transfer_one]
    simp [← coeff_zero_eq_constantCoeff, h0]
  right_inv a := by
    rw [transfer_comp, PowerSeries.mul_inv_cancel, transfer_one]
    simp [← coeff_zero_eq_constantCoeff, h0]

theorem coordinate_forward (h a : ℕ → ℚ) (h0 : h 0 = 1) (n : ℕ) :
    coordinateEquiv h h0 a n = ∑ r ∈ Finset.range (n+1), (n.choose r : ℚ)*h r* a (n-r) :=
  egf_product h a n

theorem coordinate_inverse (h a : ℕ → ℚ) (h0 : h 0 = 1) (n : ℕ) :
    (coordinateEquiv h h0).symm a n =
      ∑ r ∈ Finset.range (n+1), (n.choose r : ℚ)*decode (encode h)⁻¹ r* a (n-r) := by
  have hp := egf_product (decode (encode h)⁻¹) a n
  simpa [coordinateEquiv, transfer] using hp

theorem inverse_constant_one (h : ℕ → ℚ) (h0 : h 0 = 1) :
    decode (encode h)⁻¹ 0 = 1 := by
  simp [decode, coeff_zero_eq_constantCoeff, encode, h0]

def transportedPacket (h : ℕ → ℚ) (h0 : h 0 = 1) (N j : ℕ) (a : ℕ → ℚ) : ℕ → ℚ :=
  coordinateEquiv h h0 (ResiduePackets.packet N j ((coordinateEquiv h h0).symm a))

theorem transported_idempotent (h : ℕ → ℚ) (h0 : h 0 = 1) (N j : ℕ) (a : ℕ → ℚ) :
    transportedPacket h h0 N j (transportedPacket h h0 N j a) = transportedPacket h h0 N j a := by
  simp [transportedPacket]

theorem transported_orthogonal (h : ℕ → ℚ) (h0 : h 0 = 1) (N i j : ℕ)
    (hij : i ≠ j) (a : ℕ → ℚ) :
    transportedPacket h h0 N i (transportedPacket h h0 N j a) = 0 := by
  simp only [transportedPacket, Equiv.symm_apply_apply,
    ResiduePackets.packet_orthogonal N i j hij]
  exact transfer_zero _

theorem transported_reconstruction (h : ℕ → ℚ) (h0 : h 0 = 1) {N : ℕ}
    (hN : 0 < N) (a : ℕ → ℚ) : ∑ j ∈ Finset.range N, transportedPacket h h0 N j a = a := by
  change (∑ j ∈ Finset.range N, transfer (encode h)
    (ResiduePackets.packet N j ((coordinateEquiv h h0).symm a))) = a
  rw [← transfer_sum, ResiduePackets.packet_reconstruction hN]
  exact (coordinateEquiv h h0).apply_symm_apply a

def matrixEntry (h : ℕ → ℚ) (n q : ℕ) : ℚ :=
  if q ≤ n then (n.choose q : ℚ) * h (n-q) else 0

theorem matrix_diagonal (h : ℕ → ℚ) (h0 : h 0 = 1) (n : ℕ) : matrixEntry h n n = 1 := by
  simp [matrixEntry, h0]

theorem matrix_above_diagonal (h : ℕ → ℚ) {n q : ℕ} (hq : n < q) : matrixEntry h n q = 0 := by
  simp [matrixEntry, Nat.not_le.mpr hq]

end KomorebiPaperIX.EGF
