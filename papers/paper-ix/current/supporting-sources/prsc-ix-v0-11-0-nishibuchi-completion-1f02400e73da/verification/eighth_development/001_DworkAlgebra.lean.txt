module

public import Mathlib.RingTheory.Derivation.Basic
public import Mathlib.Data.Nat.Multiplicity
public import Mathlib.Algebra.Ring.Subring.Basic
public import Mathlib.Tactic

@[expose] public section
noncomputable section
set_option autoImplicit false

namespace KomorebiPaperIX.DworkAlgebra

variable {K A : Type*} [Field K] [CharZero K] [CommRing A] [Algebra K A]

/-- Actual powers of the underlying differentiation map. -/
def deriv (D : Derivation K A A) (n : ℕ) : A →ₗ[K] A := D.toLinearMap ^ n

@[simp] theorem deriv_zero (D : Derivation K A A) (f : A) : deriv D 0 f = f := rfl

@[simp] theorem deriv_one (D : Derivation K A A) (f : A) : deriv D 1 f = D f := by
  simp [deriv]

theorem deriv_succ (D : Derivation K A A) (n : ℕ) (f : A) :
    deriv D (n+1) f = D (deriv D n f) := by
  simp [deriv, pow_succ', Module.End.mul_apply]

theorem deriv_succ_right (D : Derivation K A A) (n : ℕ) (f : A) :
    deriv D (n+1) f = deriv D n (D f) := by
  simp [deriv, pow_succ, Module.End.mul_apply]

theorem deriv_leibniz (D : Derivation K A A) (n : ℕ) (f g : A) :
    deriv D n (f*g) = ∑ k ∈ Finset.range (n+1),
      (n.choose k : A) * deriv D k f * deriv D (n-k) g := by
  induction n with
  | zero => simp
  | succ n ih =>
    rw [deriv_succ, ih, map_sum]
    have hs := Finset.sum_choose_succ_mul
      (fun i j => deriv D i f * deriv D j g) n
    simp only [← mul_assoc] at hs
    rw [hs, ← Finset.sum_add_distrib]
    apply Finset.sum_congr rfl
    intro k hk
    have hkn : k ≤ n := Finset.mem_range_succ_iff.mp hk
    simp only [Derivation.leibniz, Derivation.map_natCast, smul_eq_mul, zero_mul,
      add_zero, deriv_succ]
    rw [show n+1-k=(n-k)+1 by omega, deriv_succ]
    ring

/-- The integral middle correction in a prime-power Leibniz row. -/
def middle (D : Derivation K A A) (p q : ℕ) (f g : A) : A :=
  ∑ k ∈ Finset.range (q-1), ((q.choose (k+1) / p : ℕ) : A) *
    deriv D (k+1) f * deriv D (q-(k+1)) g

/-- Divisibility of the binomial coefficients is proved for both prime-power rows. -/
theorem prime_power_leibniz (D : Derivation K A A) (p r : ℕ) (hp : p.Prime)
    (f g : A) :
    deriv D (p^r) (f*g) = f * deriv D (p^r) g + deriv D (p^r) f * g +
      (p:A) * middle D p (p^r) f g := by
  have hq : 0 < p^r := pow_pos hp.pos r
  rw [deriv_leibniz, Finset.sum_range_succ]
  have hsplit : (∑ k ∈ Finset.range (p^r), ((p^r).choose k : A) * deriv D k f *
      deriv D (p^r-k) g) =
      f * deriv D (p^r) g + ∑ k ∈ Finset.range (p^r-1),
        ((p^r).choose (k+1) : A) * deriv D (k+1) f * deriv D (p^r-(k+1)) g := by
    conv_lhs => rw [show p^r=(p^r-1)+1 by omega, Finset.sum_range_succ']
    simp only [Nat.sub_add_cancel (show 1 ≤ p^r by omega), Nat.choose_zero_right,
      Nat.cast_one, deriv_zero, one_mul, Nat.sub_zero]
    ring
  rw [hsplit]
  simp only [Nat.choose_self, Nat.cast_one, deriv_zero, Nat.sub_self, one_mul]
  rw [middle, Finset.mul_sum]
  have he : (∑ k ∈ Finset.range (p^r-1), ((p^r).choose (k+1) : A) *
      deriv D (k+1) f * deriv D (p^r-(k+1)) g) =
      ∑ k ∈ Finset.range (p^r-1), (p:A) *
        (((p^r).choose (k+1) / p : ℕ) : A) *
        deriv D (k+1) f * deriv D (p^r-(k+1)) g := by
    apply Finset.sum_congr rfl
    intro k hk
    have hk' : k+1 ≠ p^r := by have := Finset.mem_range.mp hk; omega
    have hd := hp.dvd_choose_pow (by omega : k+1 ≠ 0) hk'
    rw [← Nat.cast_mul, Nat.mul_div_cancel' hd]
  rw [he]
  ring

/-- Normalized differential Frobenius defect, formed in the ambient field algebra. -/
def B (D : Derivation K A A) (p : ℕ) : A →ₗ[K] A :=
  (p:K)⁻¹ • (deriv D (p^2) - deriv D p)

/-- Full characteristic-zero product identity, including the middle correction. -/
theorem B_mul (D : Derivation K A A) (p : ℕ) (hp : p.Prime) (f g : A) :
    B D p (f*g) = B D p f * g + f * B D p g +
      middle D p (p^2) f g - middle D p p f g := by
  have h0 : (p:K) ≠ 0 := Nat.cast_ne_zero.mpr hp.ne_zero
  have h1 := prime_power_leibniz D p 1 hp f g
  simp only [pow_one] at h1
  have h2 := prime_power_leibniz D p 2 hp f g
  simp only [B, LinearMap.smul_apply, LinearMap.sub_apply]
  rw [h2, h1]
  simp only [smul_sub, smul_add, smul_mul_assoc, mul_smul_comm,
    ← Algebra.smul_def, smul_smul, inv_mul_cancel₀ h0, one_smul]
  ring

end KomorebiPaperIX.DworkAlgebra
