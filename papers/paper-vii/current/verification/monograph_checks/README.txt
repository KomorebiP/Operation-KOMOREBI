PRACTICAL V2 REPLAY RECIPE
Run this Python snippet from the package root, changing suite to one of
original, corrected, constructed, models, guide-original, guide-extended or
guide-plates. It first validates the supplied recorded manifest. It derives a
fresh contract from that record's stated scope and exact execution-input list,
records current software versions, retains all declared results and caps, and
redirects the reviewed command to a new UTC-stamped output directory. The
recorded absolute Python path is replaced by the current interpreter. Existing
records, contracts and scientific inputs are not overwritten.

```python
from pathlib import Path
from datetime import datetime, timezone
import importlib.metadata as metadata
import json, platform, subprocess, sys

suite = 'original'
project = Path.cwd().resolve()
tools = project / 'verification/monograph_checks/tools'
core = {'original': 'run_original_20261009',
        'corrected': 'run_corrected_20261009',
        'constructed': 'run_constructed_20261009',
        'models': 'run_traditional_models_20261009'}
if suite in core:
    root = project
    manifest = root / 'verification/monograph_checks' / core[suite] / 'manifest.json'
else:
    assert suite in {'guide-original', 'guide-extended', 'guide-plates'}
    root = project / 'editable/visual_guide'
    manifest = root / 'checks/runs' / ('monograph-' + suite.split('-', 1)[1]) / 'manifest.json'
validator = tools / 'validate_manifest.py'
subprocess.run([sys.executable, str(validator), str(manifest), '--root', str(root)], check=True)
record = json.loads(manifest.read_text())
old = Path(record['output_directory'])
label = datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%S%fZ')
new = old.parent / ('replay-' + suite + '-' + label)
software = []
for entry in record['environment']['software']:
    name = entry['name']
    if name == 'runner-python':
        continue
    version = platform.python_version() if 'python' in name.lower() else metadata.version(name)
    software.append({'name': name, 'version': version})
contract = {'claim_id': record['claim_id'] + '_REPLAY_' + label,
            'mathematics': record['mathematics'],
            'randomness': record['randomness'],
            'execution_artifacts': record['execution_artifacts'],
            'software': software}
contract_dir = root / old.parent / 'replay-contracts'
contract_dir.mkdir(parents=True, exist_ok=True)
contract_path = contract_dir / (suite + '-' + label + '.json')
contract_path.write_text(json.dumps(contract, indent=2) + '\n')
run = record['run']
argv = [sys.executable, str(tools / 'run_experiment.py'),
        '--root', str(root), '--contract', str(contract_path),
        '--output', str(new), '--timeout', str(run['timeout_seconds']),
        '--max-output-bytes', str(run['max_output_bytes'])]
for name, key in [('max-memory-bytes', 'memory_bytes'),
                  ('max-cpu-seconds', 'cpu_seconds'),
                  ('max-cores', 'max_cores'), ('max-threads', 'max_threads')]:
    value = run['resource_limits'][key]
    if value is not None:
        argv += ['--' + name, str(value)]
for path in record['execution_artifacts']:
    argv += ['--input', path]
for path in record['declared_results']:
    argv += ['--result', str(new / Path(path).relative_to(old))]
child = [sys.executable] + [arg.replace(str(old), str(new)) for arg in record['command'][1:]]
subprocess.run(argv + ['--'] + child, cwd=root, check=True)
subprocess.run([sys.executable, str(validator), str(root / new / 'manifest.json'),
                '--root', str(root)], check=True)
```
