import PaperXII.AdicQuotientTopology
import PaperXII.FiniteTensorQuotient

/-!
# Completed tensor products of finite families and quotients

The objects are the actual Hausdorff completions of the actual finite algebraic
tensor products, with the topology defined by the base ideal and the factor ideals.
The base term includes the empty family without replacing the base-adic topology
by the discrete topology. No completeness or quotient-comparison interface is assumed.
-/
noncomputable section
open Filter Topology UniformSpace
open scoped Uniformity
set_option autoImplicit false
namespace PaperXII.FiniteTensor

variable {O ι : Type*} [CommRing O] [Fintype ι] [DecidableEq ι]
variable (R : ι → Type*) [∀ i, CommRing (R i)] [∀ i, Algebra O (R i)]
variable (mO : Ideal O) (m : ∀ i, Ideal (R i))

/-- The actual finite completed tensor product. -/
abbrev CompletedTensor := AdicModel.Completed (Raw (O := O) R) (definingIdeal R mO m)

/-- The algebra structure is induced by the canonical dense embedding. -/
instance completedTensorAlgebra : Algebra O (CompletedTensor R mO m) :=
  ((AdicModel.embed (Raw (O := O) R) (definingIdeal R mO m)).comp
    (algebraMap O (Raw (O := O) R))).toAlgebra

/-- Actual dense tensor inclusion, as an algebra homomorphism. -/
def tensorEmbed : Raw (O := O) R →ₐ[O] CompletedTensor R mO m where
  __ := AdicModel.embed (Raw (O := O) R) (definingIdeal R mO m)
  commutes' _ := rfl

/-- Actual completed factor map. -/
def completedInclude (i : ι) : R i →ₐ[O] CompletedTensor R mO m :=
  (tensorEmbed R mO m).comp (includeFactor R i)

/-- The closed ideal generated by all the supplied local condition ideals. -/
def closedConditions (J : ∀ i, Ideal (R i)) : Ideal (CompletedTensor R mO m) :=
  CompletionQuotient.closedExtension
    (AdicModel.carrierIdeal (definingIdeal R mO m) (conditions R J))

/-- This is precisely the literal closed ideal of the completed factor images. -/
omit [Fintype ι] in
theorem closedConditions_eq (J : ∀ i, Ideal (R i)) :
    closedConditions R mO m J =
      (⨆ i, (J i).map (completedInclude R mO m i).toRingHom).closure := by
  simp only [closedConditions, CompletionQuotient.closedExtension, AdicModel.carrierIdeal,
    conditions, Ideal.map_iSup, Ideal.map_map]
  rfl

instance closedConditions_isClosed (J : ∀ i, Ideal (R i)) :
    IsClosed (closedConditions R mO m J : Set (CompletedTensor R mO m)) := isClosed_closure

variable (J : ∀ i, Ideal (R i))

/-- The full actual completed tensor quotient comparison, in the construction direction. -/
def completedQuotientEquiv :
    (CompletedTensor R mO m ⧸ closedConditions R mO m J) ≃+*
      CompletedTensor (fun i ↦ R i ⧸ J i) mO
        (fun i ↦ (m i).map (Ideal.Quotient.mk (J i))) :=
  (AdicModel.quotientCompletionEquiv (definingIdeal R mO m) (conditions R J)).trans
    (AdicModel.completedEquiv _ _ (quotientEquiv R J).toRingEquiv
      (quotientEquiv_definingIdeal R J mO m))

theorem completedQuotientEquiv_continuous :
    Continuous (completedQuotientEquiv R mO m J) :=
  (AdicModel.completedEquiv_continuous _ _ (quotientEquiv R J).toRingEquiv
    (quotientEquiv_definingIdeal R J mO m)).comp
    (AdicModel.quotientCompletionEquiv_continuous (definingIdeal R mO m) (conditions R J))

theorem completedQuotientEquiv_symm_continuous :
    Continuous (completedQuotientEquiv R mO m J).symm :=
  (AdicModel.quotientCompletionEquiv_symm_continuous (definingIdeal R mO m) (conditions R J)).comp
    (AdicModel.completedEquiv_symm_continuous _ _ (quotientEquiv R J).toRingEquiv
      (quotientEquiv_definingIdeal R J mO m))

@[simp] theorem completedQuotientEquiv_mk_embed (x : Raw (O := O) R) :
    completedQuotientEquiv R mO m J (Ideal.Quotient.mk _ (tensorEmbed R mO m x)) =
      tensorEmbed (fun i ↦ R i ⧸ J i) mO
        (fun i ↦ (m i).map (Ideal.Quotient.mk (J i))) (quotientMap R J x) := by
  change AdicModel.completedEquiv _ _ (quotientEquiv R J).toRingEquiv
    (quotientEquiv_definingIdeal R J mO m)
      (AdicModel.quotientCompletionEquiv _ _ (Ideal.Quotient.mk _ (AdicModel.embed _ _ x))) = _
  rw [AdicModel.quotientCompletionEquiv_mk_embed, AdicModel.completedEquiv_embed]
  rfl

/-- Claim 3.1 with the source's displayed direction and its coefficient-algebra structure. -/
def completed_tensor_quotient :
    CompletedTensor (fun i ↦ R i ⧸ J i) mO
      (fun i ↦ (m i).map (Ideal.Quotient.mk (J i))) ≃ₐ[O]
      (CompletedTensor R mO m ⧸ closedConditions R mO m J) :=
  (show (CompletedTensor R mO m ⧸ closedConditions R mO m J) ≃ₐ[O]
      CompletedTensor (fun i ↦ R i ⧸ J i) mO
        (fun i ↦ (m i).map (Ideal.Quotient.mk (J i))) from
    { __ := completedQuotientEquiv R mO m J
      commutes' r := by
        change completedQuotientEquiv R mO m J
          (Ideal.Quotient.mk _ (tensorEmbed R mO m (algebraMap O (Raw (O := O) R) r))) = _
        rw [completedQuotientEquiv_mk_embed, (quotientMap R J).commutes]
        rfl }).symm

/-- Both maps of the completed tensor quotient equivalence are continuous. -/
theorem completed_tensor_quotient_continuous :
    Continuous (completed_tensor_quotient R mO m J) ∧
      Continuous (completed_tensor_quotient R mO m J).symm :=
  ⟨completedQuotientEquiv_symm_continuous R mO m J,
    completedQuotientEquiv_continuous R mO m J⟩

/-- The isomorphism respects every actual canonical factor map. -/
@[simp] theorem completed_tensor_quotient_factor (i : ι) (x : R i) :
    completed_tensor_quotient R mO m J
      (completedInclude (fun i ↦ R i ⧸ J i) mO
        (fun i ↦ (m i).map (Ideal.Quotient.mk (J i))) i (Ideal.Quotient.mk (J i) x)) =
      Ideal.Quotient.mk _ (completedInclude R mO m i x) := by
  apply (completedQuotientEquiv R mO m J).injective
  change completedQuotientEquiv R mO m J
    ((completedQuotientEquiv R mO m J).symm _) = _
  rw [RingEquiv.apply_symm_apply]
  symm
  simpa only [quotientMap_include] using
    completedQuotientEquiv_mk_embed R mO m J (includeFactor R i x)

attribute [local instance] CompletionQuotient.quotientUniformSpace

/-- The right-hand quotient is itself a genuine complete Hausdorff ring. -/
omit [Fintype ι] in
theorem completed_tensor_quotient_complete :
    CompleteSpace (CompletedTensor R mO m ⧸ closedConditions R mO m J) ∧
      T2Space (CompletedTensor R mO m ⧸ closedConditions R mO m J) :=
  ⟨CompletionQuotient.quotient_complete _, inferInstance⟩

end PaperXII.FiniteTensor
