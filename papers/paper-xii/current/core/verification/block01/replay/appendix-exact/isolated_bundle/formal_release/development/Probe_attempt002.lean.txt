example {α : Type*} [UniformSpace α] [Filter.IsCountablyGenerated (𝓤 α)] :
 Filter.IsCountablyGenerated (𝓤 (CauchyFilter α)) := by
 obtain ⟨s,hs⟩ := Filter.isCountablyGenerated_iff_exists_antitone_basis.mp
   (inferInstance : Filter.IsCountablyGenerated (𝓤 α))
 exact (CauchyFilter.basis_uniformity hs.toHasBasis).isCountablyGenerated
