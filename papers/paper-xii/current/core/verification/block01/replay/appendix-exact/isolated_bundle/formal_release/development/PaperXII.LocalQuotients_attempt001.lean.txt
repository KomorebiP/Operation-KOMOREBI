import Mathlib.Topology.Algebra.Ring.Ideal
import Mathlib.RingTheory.LocalRing.RingHom.Basic
import Mathlib.RingTheory.Ideal.Maps

/-! Actual quotient factorization, including continuity and the closed ideal
generated by local conditions. This proves the factorization assertion of
Paper XII 3.2 for the supplied topological rings and inclusions. -/
set_option autoImplicit false
namespace PaperXII.LocalQuotients

variable {R A : Type*} [CommRing R] [CommRing A]

/-- Factorization of the original point through the actual ideal quotient. -/
def Factors (J : Ideal R) (f : R →+* A) : Prop :=
  ∃ g : R ⧸ J →+* A, g.comp (Ideal.Quotient.mk J) = f

/-- Algebraic factorization is exactly kernel containment. -/
theorem factors_iff (J : Ideal R) (f : R →+* A) :
    Factors J f ↔ J ≤ RingHom.ker f := by
  constructor
  · rintro ⟨g, hg⟩ a ha
    change f a = 0
    rw [← hg]
    change g (Ideal.Quotient.mk J a) = 0
    rw [Ideal.Quotient.eq_zero_iff_mem.mpr ha, map_zero]
  · intro h
    exact ⟨Ideal.Quotient.lift J f h, by ext a; rfl⟩

/-- The quotient map is surjective, so its factor is unique. -/
theorem factor_unique (J : Ideal R) (f : R →+* A) (g h : R ⧸ J →+* A)
    (hg : g.comp (Ideal.Quotient.mk J) = f)
    (hh : h.comp (Ideal.Quotient.mk J) = f) : g = h :=
  Ideal.Quotient.ringHom_ext (hg.trans hh.symm)

/-- A factor of a local point is itself a local map. -/
theorem factor_isLocal (J : Ideal R) (f : R →+* A) [IsLocalHom f]
    (g : R ⧸ J →+* A) (hg : g.comp (Ideal.Quotient.mk J) = f) : IsLocalHom g := by
  constructor
  intro x hx
  obtain ⟨a, rfl⟩ := Ideal.Quotient.mk_surjective x
  have ha : IsUnit (f a) := by
    rw [← hg]
    exact hx
  exact (isUnit_of_map_unit f a ha).map (Ideal.Quotient.mk J)

variable [TopologicalSpace R] [IsTopologicalRing R] [TopologicalSpace A]

/-- Continuous factorization uses the actual quotient topology. -/
def ContinuousFactors (J : Ideal R) (f : R →+* A) : Prop :=
  ∃ g : R ⧸ J →+* A, Continuous g ∧ g.comp (Ideal.Quotient.mk J) = f

/-- The actual quotient factor of a continuous point is continuous. -/
theorem continuous_lift (J : Ideal R) (f : R →+* A) (hf : Continuous f)
    (hJ : J ≤ RingHom.ker f) : Continuous (Ideal.Quotient.lift J f hJ) := by
  apply (QuotientRing.isOpenQuotientMap_mk J).isQuotientMap.continuous_iff.mpr
  exact hf

/-- There is no additional continuity obstruction after kernel containment. -/
theorem continuous_factors_iff (J : Ideal R) (f : R →+* A) (hf : Continuous f) :
    ContinuousFactors J f ↔ J ≤ RingHom.ker f := by
  constructor
  · rintro ⟨g, _, hg⟩
    exact (factors_iff J f).mp ⟨g, hg⟩
  · intro hJ
    exact ⟨Ideal.Quotient.lift J f hJ, continuous_lift J f hf hJ, by ext a; rfl⟩

/-- Existence and uniqueness hold for the continuous quotient factor of the same point. -/
theorem exists_unique_continuous_factor (J : Ideal R) (f : R →+* A)
    (hf : Continuous f) (hJ : J ≤ RingHom.ker f) :
    ∃! g : R ⧸ J →+* A, Continuous g ∧ g.comp (Ideal.Quotient.mk J) = f := by
  obtain ⟨g, hc, hg⟩ := (continuous_factors_iff J f hf).mpr hJ
  exact ⟨g, ⟨hc, hg⟩, fun h hh ↦ factor_unique J f h g hh.2 hg⟩

variable [T2Space A]

omit [IsTopologicalRing R] in
/-- The Hausdorff target makes the kernel of a continuous point closed. -/
theorem kernel_closed (f : R →+* A) (hf : Continuous f) :
    IsClosed (RingHom.ker f : Set R) := isClosed_eq hf continuous_const

/-- Killing an ideal and killing its topological closure are equivalent. -/
theorem closure_le_kernel_iff (J : Ideal R) (f : R →+* A) (hf : Continuous f) :
    J.closure ≤ RingHom.ker f ↔ J ≤ RingHom.ker f := by
  constructor
  · intro h a ha
    exact h (subset_closure ha)
  · intro h
    exact closure_minimal h (kernel_closed f hf)

variable {V : Type*} (S : V → Type*) [∀ v, CommRing (S v)]
    (inc : ∀ v, S v →+* R) (J : ∀ v, Ideal (S v))

/-- The literal closed ideal generated by all local condition ideals. -/
def closedConditions : Ideal R := (⨆ v, (J v).map (inc v)).closure

/-- The closure step and restriction to every local ideal are both proved. -/
theorem closed_conditions_iff (f : R →+* A) (hf : Continuous f) :
    closedConditions S inc J ≤ RingHom.ker f ↔
      ∀ v, J v ≤ RingHom.ker (f.comp (inc v)) := by
  rw [closedConditions, closure_le_kernel_iff _ f hf, iSup_le_iff]
  apply forall_congr'
  intro v
  rw [Ideal.map_le_iff_le_comap]
  rfl

/-- Corollary 3.2: the global point factors exactly when all local restrictions do. -/
theorem factors_iff_local_factors [∀ v, TopologicalSpace (S v)]
    [∀ v, IsTopologicalRing (S v)] (hinc : ∀ v, Continuous (inc v))
    (f : R →+* A) (hf : Continuous f) :
    ContinuousFactors (closedConditions S inc J) f ↔
      ∀ v, ContinuousFactors (J v) (f.comp (inc v)) := by
  rw [continuous_factors_iff _ f hf, closed_conditions_iff S inc J f hf]
  exact forall_congr' (fun v ↦ (continuous_factors_iff (J v)
    (f.comp (inc v)) (hf.comp (hinc v))).symm)

/-- When the point is local, its unique continuous global quotient factor is local. -/
theorem exists_unique_local_factor (f : R →+* A) [IsLocalHom f]
    (hf : Continuous f) (h : ∀ v, J v ≤ RingHom.ker (f.comp (inc v))) :
    ∃! g : R ⧸ closedConditions S inc J →+* A,
      Continuous g ∧ IsLocalHom g ∧ g.comp (Ideal.Quotient.mk _) = f := by
  obtain ⟨g, hg, _⟩ := exists_unique_continuous_factor (closedConditions S inc J) f hf
    ((closed_conditions_iff S inc J f hf).mpr h)
  exact ⟨g, ⟨hg.1, factor_isLocal _ f g hg.2, hg.2⟩,
    fun k hk ↦ factor_unique _ f k g hk.2.2 hg.2⟩

end PaperXII.LocalQuotients
