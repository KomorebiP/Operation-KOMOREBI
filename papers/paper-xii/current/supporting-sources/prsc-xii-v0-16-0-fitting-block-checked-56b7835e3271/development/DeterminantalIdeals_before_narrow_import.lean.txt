import Mathlib.LinearAlgebra.Matrix.Determinant.Basic
import Mathlib.RingTheory.Ideal.Maps
import Mathlib.Tactic

/-!
# Determinantal ideals for Paper XII

These are ideals of actual minors, not supplied invariant ideals. Repeated row or
column indices are permitted in the generating family; their determinants vanish.
This module is infrastructure, not yet the presentation-independence theorem.
-/
noncomputable section
namespace PaperXII.DeterminantalIdeals
open Matrix
variable {R S : Type*} [CommRing R] [CommRing S]
variable {m n p : Type*}

/-- Ideal generated by all size-`k` minors. The empty minor equals one. -/
def minorIdeal (A : Matrix m n R) (k : ℕ) : Ideal R :=
  Ideal.span (Set.range fun rc : (Fin k → m) × (Fin k → n) =>
    (A.submatrix rc.1 rc.2).det)

theorem minor_mem (A : Matrix m n R) (k : ℕ)
    (r : Fin k → m) (c : Fin k → n) :
    (A.submatrix r c).det ∈ minorIdeal A k :=
  Ideal.subset_span ⟨(r,c), rfl⟩

theorem minorIdeal_le_iff (A : Matrix m n R) (k : ℕ) (J : Ideal R) :
    minorIdeal A k ≤ J ↔ ∀ (r : Fin k → m) (c : Fin k → n), (A.submatrix r c).det ∈ J := by
  rw [minorIdeal, Ideal.span_le]
  constructor
  · intro h r c
    exact h ⟨(r,c),rfl⟩
  · rintro h _ ⟨⟨r,c⟩,rfl⟩
    exact h r c

@[simp] theorem minorIdeal_zero (A : Matrix m n R) : minorIdeal A 0 = ⊤ := by
  apply (Ideal.eq_top_iff_one _).mpr
  simpa using minor_mem A 0 Fin.elim0 Fin.elim0

/-- Minors commute with an arbitrary coefficient-ring map. No flatness is used. -/
theorem minorIdeal_map (f : R →+* S) (A : Matrix m n R) (k : ℕ) :
    minorIdeal (A.map f) k = Ideal.map f (minorIdeal A k) := by
  rw [minorIdeal, minorIdeal, Ideal.map_span]
  congr 1
  ext x
  constructor
  · rintro ⟨⟨r,c⟩,rfl⟩
    refine ⟨(A.submatrix r c).det, ⟨(r,c),rfl⟩, ?_⟩
    exact f.map_det _
  · rintro ⟨_, ⟨⟨r,c⟩,rfl⟩,rfl⟩
    exact ⟨(r,c), (f.map_det _).symm⟩

@[simp] theorem minorIdeal_transpose (A : Matrix m n R) (k : ℕ) :
    minorIdeal A.transpose k = minorIdeal A k := by
  apply le_antisymm
  · apply (minorIdeal_le_iff _ _ _).2
    intro r c
    simpa only [← Matrix.transpose_submatrix, Matrix.det_transpose] using minor_mem A k c r
  · apply (minorIdeal_le_iff _ _ _).2
    intro r c
    simpa only [← Matrix.transpose_submatrix, Matrix.det_transpose,
      Matrix.transpose_transpose] using minor_mem A.transpose k c r

/-- Laplace expansion gives the descending determinantal filtration. -/
theorem minorIdeal_succ_le (A : Matrix m n R) (k : ℕ) :
    minorIdeal A (k+1) ≤ minorIdeal A k := by
  apply (minorIdeal_le_iff _ _ _).2
  intro r c
  rw [Matrix.det_succ_row _ 0]
  apply Ideal.sum_mem
  intro j _
  exact Ideal.mul_mem_left _ _ (minor_mem A k (r ∘ (0 : Fin (k+1)).succAbove)
    (c ∘ j.succAbove))

/-- Rectangular matrix multiplication cannot enlarge the minors ideal of the right factor. -/
theorem minorIdeal_mul_le_right [Fintype n] (A : Matrix m n R) (B : Matrix n p R)
    (k : ℕ) : minorIdeal (A * B) k ≤ minorIdeal B k := by
  apply (minorIdeal_le_iff _ _ _).2
  intro r c
  have hrows : (A * B).submatrix r c =
      fun i => ∑ t : n, A (r i) t • (fun j => B t (c j)) := by
    ext i j
    simp [Matrix.mul_apply, Matrix.submatrix_apply, Pi.smul_apply, Finset.sum_apply]
  rw [hrows]
  change (Matrix.detRowAlternating (R := R) (n := Fin k)).toMultilinearMap
    (fun i => ∑ t : n, A (r i) t • (fun j => B t (c j))) ∈ minorIdeal B k
  rw [MultilinearMap.map_sum]
  apply Ideal.sum_mem
  intro t _
  change Matrix.detRowAlternating (fun i => A (r i) (t i) • (fun j => B (t i) (c j))) ∈ minorIdeal B k
  rw [Matrix.detRowAlternating.map_smul_univ]
  exact (minorIdeal B k).smul_mem _ (minor_mem B k t c)

/-- The analogous left-factor inclusion. -/
theorem minorIdeal_mul_le_left [Fintype n] (A : Matrix m n R) (B : Matrix n p R)
    (k : ℕ) : minorIdeal (A * B) k ≤ minorIdeal A k := by
  have h := minorIdeal_mul_le_right B.transpose A.transpose k
  simpa only [← Matrix.transpose_mul, minorIdeal_transpose] using h

/-- An actual two-sided inverse gives invariance, not merely containment. -/
theorem minorIdeal_mul_left_eq [Fintype m] [DecidableEq m] (U V : Matrix m m R)
    (hVU : V * U = 1) (A : Matrix m n R) (k : ℕ) :
    minorIdeal (U * A) k = minorIdeal A k := by
  classical
  apply le_antisymm (minorIdeal_mul_le_right U A k)
  have h := minorIdeal_mul_le_right V (U*A) k
  simpa only [← Matrix.mul_assoc, hVU, Matrix.one_mul] using h

theorem minorIdeal_mul_right_eq [Fintype n] [DecidableEq n] (U V : Matrix n n R)
    (hUV : U * V = 1) (A : Matrix m n R) (k : ℕ) :
    minorIdeal (A * U) k = minorIdeal A k := by
  classical
  apply le_antisymm (minorIdeal_mul_le_left A U k)
  have h := minorIdeal_mul_le_left (A*U) V k
  simpa only [Matrix.mul_assoc, hUV, Matrix.mul_one] using h

/-- Adding or removing redundant relation columns is harmless if both matrices factor. -/
theorem minorIdeal_eq_of_mutual_factorization [Fintype n] [Fintype p]
    (A : Matrix m n R) (B : Matrix m p R)
    (C : Matrix p n R) (D : Matrix n p R) (hA : B*C=A) (hB : A*D=B) (k : ℕ) :
    minorIdeal A k = minorIdeal B k := by
  apply le_antisymm
  · rw [← hA]
    exact minorIdeal_mul_le_left B C k
  · rw [← hB]
    exact minorIdeal_mul_le_left A D k

end PaperXII.DeterminantalIdeals
