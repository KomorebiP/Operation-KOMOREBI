theorem sum_blocks (A : M →ₗ[R] M) :
∑mu, ∑nu, P.block A mu nu = A := by
have hcomplete (x : M) : ∑i, P.projector i x = x := by
have hx := congrArg (fun L : M →ₗ[R] M ↦L x) P.complete
