   13 /-- An actual analytic logarithm on a stated domain. -/
   14 def IsLogarithmOn (f L : ℂ → ℂ) (s : Set ℂ) : Prop :=
   15   DifferentiableOn ℂ L s ∧ ∀ z ∈ s, Complex.exp (L z) = f z

   17 /-- Reciprocal powers using one fixed logarithm on the whole domain. -/
   18 def logarithmPower (L : ℂ → ℂ) (r : ℂ) (z : ℂ) : ℂ :=
   19   Complex.exp (-r * L z)
   20 
   21 theorem logarithmPower_add (L : ℂ → ℂ) (r t z : ℂ) :
   22     logarithmPower L r z * logarithmPower L t z = logarithmPower L (r + t) z := by
   23   unfold logarithmPower
   24   rw [← Complex.exp_add]
   25   congr 1
   26   ring
   27 
   28 theorem IsLogarithmOn.nonzero {f L : ℂ → ℂ} {s : Set ℂ}
   29     (hL : IsLogarithmOn f L s) {z : ℂ} (hz : z ∈ s) : f z ≠ 0 := by
   30   rw [← hL.2 z hz]
   31   exact Complex.exp_ne_zero _
   32 
   33 theorem IsLogarithmOn.deriv {f L : ℂ → ℂ} {s : Set ℂ}
   34     (hs : IsOpen s) (hL : IsLogarithmOn f L s) {z : ℂ} (hz : z ∈ s) :
   35     deriv L z = logDeriv f z := by
   36   have hd := (hL.1 z hz).differentiableAt (hs.mem_nhds hz)
   37   have he : (fun w => Complex.exp (L w)) =ᶠ[𝓝 z] f :=
   38     eventually_of_mem (hs.mem_nhds hz) (fun w hw => hL.2 w hw)
   39   have h := he.deriv_eq
   40   rw [hd.hasDerivAt.cexp.deriv, hL.2 z hz] at h
   41   rw [logDeriv_apply]
   42   apply (eq_div_iff (hL.nonzero hz)).mpr
   43   simpa [mul_comm] using h

   45 theorem hasDerivAt_logarithmPower {f L : ℂ → ℂ} {s : Set ℂ}
   46     (hs : IsOpen s) (hL : IsLogarithmOn f L s) (r : ℂ) {z : ℂ} (hz : z ∈ s) :
   47     HasDerivAt (logarithmPower L r)
   48       (r * (-logDeriv f z) * logarithmPower L r z) z := by

   53   rw [hL.deriv hs hz] at h
   54   simpa only [logarithmPower, neg_mul, mul_neg, mul_assoc, mul_left_comm, mul_comm] using h

   63 /-- Evaluation of the complete differential Bell polynomial on successive
   64 derivatives of a function. Its defining recurrence is Y_(n+1)=Y_n'+gY_n. -/
   65 def analyticDifferentialBell (g : ℂ → ℂ) : ℕ → ℂ → ℂ
   66   | 0 => fun _ => 1
   67   | n + 1 => fun z => deriv (analyticDifferentialBell g n) z + g z * analyticDifferentialBell g n z
   68 
   69 theorem analyticOnNhd_differentialBell {g : ℂ → ℂ} {s : Set ℂ}
   70     (hg : AnalyticOnNhd ℂ g s) (n : ℕ) : AnalyticOnNhd ℂ (analyticDifferentialBell g n) s := by
   71   induction n with
   72   | zero => intro z _; exact analyticAt_const
   73   | succ n ih => exact ih.deriv.add (hg.mul ih)
   74 
   75 theorem iteratedDeriv_eq_analyticDifferentialBell {f g : ℂ → ℂ} {s : Set ℂ}
   76     (hs : IsOpen s) (hf : AnalyticOnNhd ℂ f s) (hg : AnalyticOnNhd ℂ g s)
   77     (hfg : ∀ z ∈ s, deriv f z = g z * f z) (n : ℕ) :
   78     ∀ z ∈ s, iteratedDeriv n f z = analyticDifferentialBell g n z * f z := by
   79   induction n with
   80   | zero => intro z _; simp [analyticDifferentialBell]
   81   | succ n ih =>
   82     intro z hz
   83     have he : iteratedDeriv n f =ᶠ[𝓝 z]
   84         (fun w => analyticDifferentialBell g n w * f w) :=
   85       eventually_of_mem (hs.mem_nhds hz) (fun w hw => ih w hw)
   86     rw [iteratedDeriv_succ, he.deriv_eq,
   87       ((analyticOnNhd_differentialBell hg n z hz).differentiableAt.hasDerivAt.fun_mul
   88         (hf z hz).differentiableAt.hasDerivAt).deriv,
   89       hfg z hz]
   90     simp only [analyticDifferentialBell]
   91     ring
   92 
   93 theorem analytic_higher_riccati {N : ℕ} (hN : 0 < N) {z : ℂ}
   94     (hz : canonicalSignedDenominator N z ≠ 0) :
   95     analyticDifferentialBell (fun w => -analyticTransferTangent N w) N z = -1 := by
   96   let f := canonicalSignedDenominator N
   97   let s := {w : ℂ | f w ≠ 0}
   98   have hf : Differentiable ℂ f := fun w =>
   99     (analyticAt_entireSigned N 0 (canonicalPrimitiveRoot N) (canonicalSignedRoot N) w).differentiableAt
  100   have hs : IsOpen s := isOpen_ne_fun hf.continuous continuous_const
  101   have hg : AnalyticOnNhd ℂ (fun w => -analyticTransferTangent N w) s := by
  102     intro w hw
  103     change AnalyticAt ℂ (fun w => -(-logDeriv f w)) w
  104     simp only [neg_neg, logDeriv_apply]
  105     have hfa := hf.analyticAt w
  106     have hfd := hfa.deriv
  107     exact hfd.fun_div hfa (show f w ≠ 0 from hw)
  108   have hfg : ∀ w ∈ s, deriv f w = (-analyticTransferTangent N w) * f w := by
  109     intro w hw
  110     change deriv f w = -(-logDeriv f w) * f w
  111     simp only [neg_neg, logDeriv_apply]
  112     exact (div_mul_cancel₀ _ hw).symm
  113   have he := iteratedDeriv_eq_analyticDifferentialBell hs (fun w _ => hf.analyticAt w) hg hfg N z hz
  114   have hode := entireSigned_ode (canonicalPrimitiveRoot_isPrimitive hN) (canonicalSignedRoot_pow hN) 0 z
  115   change deriv^[N] f z = -f z at hode
  116   rw [iteratedDeriv_eq_iterate, hode] at he
  117   apply mul_right_cancel₀ hz
  118   calc
  119     _ = -f z := he.symm
  120     _ = _ := by ring

   14 /-- Newton's lower Hessenberg matrix, including its negative superdiagonal. -/
   15 def newtonHessenberg (p : ℕ → K) (n : ℕ) : Matrix (Fin n) (Fin n) K :=
   16   fun i j => (if j.val ≤ i.val then p (i.val+1-j.val) else 0) -
   17     (if j.val = i.val+1 then (i.val+1 : K) else 0)
   18 
   19 omit [CharZero K] in
   20 private theorem newton_column_sum (p b : ℕ → K) (n : ℕ)
   21     (hb : ∀ i, ∑ j ∈ range (i+1), b j * p (i+1-j) = (i+1 : K)*b (i+1))
   22     (i : Fin (n+1)) :
   23     (∑ j : Fin (n+1), b j.val * newtonHessenberg p (n+1) i j) =
   24       if i = Fin.last n then (n+1 : K)*b (n+1) else 0 := by
   25   have hi : i.val+1 ≤ n+1 := i.isLt
   26   simp only [newtonHessenberg, mul_sub, sum_sub_distrib]
   27   rw [Fin.sum_univ_eq_sum_range (f := fun j => b j * (if j ≤ i.val then p (i.val+1-j) else 0)),
   28     Fin.sum_univ_eq_sum_range (f := fun j => b j * (if j = i.val+1 then (i.val+1 : K) else 0))]
   29   have hl : (∑ j ∈ range (n+1), b j * (if j ≤ i.val then p (i.val+1-j) else 0)) =
   30       ∑ j ∈ range (i.val+1), b j*p (i.val+1-j) := by
   31     symm
   32     calc
   33       _ = ∑ j ∈ range (i.val+1), b j * (if j ≤ i.val then p (i.val+1-j) else 0) := by
   34         apply sum_congr rfl
   35         intro j hj
   36         simp only [show j ≤ i.val by have := mem_range.mp hj; omega, ite_true]
   37       _ = _ := sum_subset (range_mono hi) (by
   38         intro j _ hj
   39         simp only [show ¬ j ≤ i.val by simpa using hj, ite_false, mul_zero])
   40   rw [hl, hb]
   41   by_cases hiLast : i = Fin.last n
   42   · subst i
   43     simp only [Fin.val_last, ite_true]
   44     have hz : (∑ j ∈ range (n+1), b j * (if j = n+1 then (n+1 : K) else 0)) = 0 := by
   45       apply sum_eq_zero
   46       intro j hj
   47       simp only [show j ≠ n+1 by have := mem_range.mp hj; omega, ite_false, mul_zero]
   48     rw [hz, sub_zero]
   49   · have hil : i.val+1 < n+1 := by
   50       have hn : i.val ≠ n := by intro h; apply hiLast; exact Fin.ext h
   51       omega
   52     rw [sum_eq_single (i.val+1)]
   53     · simp [hiLast, mul_comm]
   54     · intro j _ hj
   55       simp only [hj, ite_false, mul_zero]
   56     · exact fun h => (h (mem_range.mpr hil)).elim

   58 /-- The exact determinant solution of Newton's recurrence. -/
   59 theorem newtonHessenberg_det (p b : ℕ → K) (hb0 : b 0 = 1)
   60     (hb : ∀ i, ∑ j ∈ range (i+1), b j*p (i+1-j) = (i+1 : K)*b (i+1))
   61     (n : ℕ) : (newtonHessenberg p n).det = (n.factorial : K)*b n := by
   62   classical
   63   cases n with
   64   | zero => simp [hb0, Matrix.det_isEmpty]
   65   | succ n =>
   66     let H := newtonHessenberg p (n+1)
   67     let C := H.updateCol 0 (fun i => if i = Fin.last n then (n+1 : K)*b (n+1) else 0)
   68     have hc : C.det = H.det := by
   69       have h := Matrix.det_updateCol_sum H 0 (fun j => b j.val)
   70       simp only [Fin.val_zero, hb0, smul_eq_mul, one_mul] at h
   71       have he : (fun i => if i = Fin.last n then (n+1 : K)*b (n+1) else 0) =
   72           (fun i => ∑ j, b j.val * H i j) := by
   73         funext i
   74         exact (newton_column_sum p b n hb i).symm
   75       change (H.updateCol 0 _).det = H.det
   76       rw [he]
   77       exact h
   78     have hm : C.submatrix (Fin.last n).succAbove Fin.succ =
   79         fun i j : Fin n => newtonHessenberg p (n+1) i.castSucc j.succ := by
   80       ext i j
   81       simp [C, H, Matrix.submatrix, Matrix.updateCol, Fin.succ_ne_zero]
   82     have ht : (C.submatrix (Fin.last n).succAbove Fin.succ).IsLowerTriangular := by
   83       rw [hm]
   84       intro i j hij
   85       simp only [newtonHessenberg, Fin.val_castSucc, Fin.val_succ]
   86       have hij' : i.val < j.val := hij
   87       simp [show ¬ j.val+1 ≤ i.val by omega, show j.val ≠ i.val by omega]
   88     have hd : (C.submatrix (Fin.last n).succAbove Fin.succ).det =
   89         (-1 : K)^n * (n.factorial : K) := by
   90       rw [Matrix.det_of_isLowerTriangular _ ht, hm]
   91       simp only [newtonHessenberg, Fin.val_castSucc, Fin.val_succ,
   92         Nat.not_succ_le_self, ite_false, ite_true, zero_sub]
   93       rw [Finset.prod_neg]
   94       simp only [card_univ, Fintype.card_fin]
   95       congr 1
   96       rw [Fin.prod_univ_eq_prod_range (f := fun j => (j : K)+1), Nat.factorial_eq_prod_range_add_one, Nat.cast_prod]
   97       apply prod_congr rfl
   98       intro j _
   99       simp
  100     rw [← hc, Matrix.det_succ_column_zero, sum_eq_single (Fin.last n)]
  101     · simp only [C, Matrix.updateCol_self, Fin.val_last, ite_true]
  102       rw [hd, Nat.factorial_succ, Nat.cast_mul, Nat.cast_add, Nat.cast_one]
  103       have hs : (-1 : K)^n * (-1)^n = 1 := by rw [← mul_pow]; simp
  104       calc
  105         _ = ((-1 : K)^n * (-1)^n) * ((n+1 : K)*(n.factorial : K)*b (n+1)) := by ring
  106         _ = _ := by rw [hs, one_mul]
  107     · intro i _ hi
  108       simp [C, hi]
  109     · simp

   76     HasSum (fun q => (residueZeroFCoeff N j q : ℂ) * w^q)
   77       (residueHypergeometric N j w) := by

   98 /-- The pointwise hypergeometric formula, with ordinary normalization and exactly N-1 parameters. -/
   99 theorem entireTransfer_hypergeometric {N j : ℕ} (hN : 0 < N) (hj : j < N)
  100     {omega : ℂ} (ho : IsPrimitiveRoot omega N) (lambda z : ℂ) :
  101     entireTransfer N j omega lambda z =
  102       (lambda*z)^j / (j.factorial : ℂ) *
  103         residueHypergeometric N j ((lambda*z)^N / (N : ℂ)^N) := by
  104   let f := fun m => PowerSeries.coeff m (transferSection N j lambda) * z^m
  105   have hi : Function.Injective (fun q : ℕ => N*q+j) := by
  106     intro a b h
  107     exact Nat.eq_of_mul_eq_mul_left hN (Nat.add_right_cancel h)
  108   have hf : ∀ m, m ∉ Set.range (fun q : ℕ => N*q+j) → f m = 0 := by
  109     intro m hm
  110     have he : m%N ≠ j := by
  111       intro he
  112       apply hm
  113       use m/N
  114       change N*(m/N)+j=m
  115       rw [← he, Nat.add_comm, Nat.mod_add_div]
  116     simp [f, coeff_transferSection, transferSectionCoeff, he]
  117   have ha := (hi.hasSum_iff hf).2 (hasSum_entireTransfer hN hj ho lambda z)
  118   have hh := (residueHypergeometric_hasSum hN hj ((lambda*z)^N / (N : ℂ)^N)).mul_left
  119     ((lambda*z)^j / (j.factorial : ℂ))
  120   apply ha.unique
  121   apply hh.congr_fun
  122   intro q
  123   dsimp [Function.comp_def, f]
  124   rw [coeff_transferSection, transferSectionCoeff,
  125     show (N*q+j)%N = j by simp [Nat.add_mod, Nat.mod_eq_of_lt hj]]
  126   simp only [one_div, map_inv₀, map_natCast]
  127   have hn := congrArg (fun a : ℚ => (a : ℂ)) (residueZeroFCoeff_normalization hN j q)
  128   push_cast at hn
  129   rw [← hn]
  130   simp only [div_eq_mul_inv, mul_pow, inv_pow, ← pow_mul]
  131   rw [pow_add, pow_add]
  132   simp only [ite_true]
  133   ring
  134 
  135 end

   14 the zero-power bounds, product estimate and Jensen-Poisson estimate prove it. -/

   16     canonicalProductQuotient (entireReducedDenominator N) = 1 := by

   23   have hgz : g = 0 := entire_sublinear_factor_rigidity hf hP hg
   24     (fullCanonicalProduct_zero _) hg0 (fun z => by rw [hgexp]; exact he z)
   25     hC (by norm_num : (3 / 4 : ℝ) < 1) hbound
   26   funext z
   27   rw [← hgexp z, hgz]
   28   exact Complex.exp_zero
