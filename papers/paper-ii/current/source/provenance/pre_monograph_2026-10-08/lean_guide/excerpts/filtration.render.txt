structure MultiplicativeFiltration (R : Type u) [Ring R] where
  Mem : ℕ → R → Prop
  zero_mem : ∀ n, Mem n 0
  add_mem : ∀ \{n x y\}, Mem n x → Mem n y → Mem n (x + y)
  neg_mem : ∀ \{n x\}, Mem n x → Mem n (-x)
  one_mem : Mem 0 1
  antitone : ∀ \{m n x\}, m ≤ n → Mem n x → Mem m x
  mul_mem : ∀ \{m n x y\}, Mem m x → Mem n y → Mem (m + n) (x * y)
