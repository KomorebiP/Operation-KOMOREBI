import Mathlib.RingTheory.PiTensorProduct
import Mathlib.RingTheory.Ideal.Quotient.Operations
import Mathlib.RingTheory.Ideal.Maps

/-! Actual finite algebra tensor products and simultaneous ideal quotients. -/
noncomputable section
open Function
set_option autoImplicit false
namespace PaperXII.FiniteTensor

variable {O ι : Type*} [CommRing O] [Fintype ι] [DecidableEq ι]
variable (R : ι → Type*) [∀ i, CommRing (R i)] [∀ i, Algebra O (R i)]

abbrev Raw := PiTensorProduct O R

/-- The canonical inclusion of one factor in the actual finite tensor product. -/
def includeFactor (i : ι) : R i →ₐ[O] Raw (O := O) R := PiTensorProduct.singleAlgHom i

/-- The universal algebra map, constructed by the multilinear product of the factor maps. -/
def liftFamily {A : Type*} [CommRing A] [Algebra O A] (f : ∀ i, R i →ₐ[O] A) :
    Raw (O := O) R →ₐ[O] A :=
  PiTensorProduct.liftAlgHom
    ((MultilinearMap.mkPiAlgebra O ι A).compLinearMap (fun i ↦ (f i).toLinearMap))
    (by simp [MultilinearMap.compLinearMap_apply, MultilinearMap.mkPiAlgebra_apply])
    (by intro x y; simp [MultilinearMap.compLinearMap_apply, MultilinearMap.mkPiAlgebra_apply,
      Pi.mul_apply, Finset.prod_mul_distrib])

omit [DecidableEq ι] in
@[simp] theorem liftFamily_tprod {A : Type*} [CommRing A] [Algebra O A]
    (f : ∀ i, R i →ₐ[O] A) (x : ∀ i, R i) :
    liftFamily R f (PiTensorProduct.tprod O x) = ∏ i, f i (x i) := by
  change PiTensorProduct.lift _ (PiTensorProduct.tprod O x) = _
  rw [PiTensorProduct.lift.tprod]
  rfl

@[simp] theorem liftFamily_include {A : Type*} [CommRing A] [Algebra O A]
    (f : ∀ i, R i →ₐ[O] A) (i : ι) (x : R i) :
    liftFamily R f (includeFactor R i x) = f i x := by
  change liftFamily R f (PiTensorProduct.tprod O (Pi.mulSingle i x)) = _
  rw [liftFamily_tprod]
  have h : (fun j ↦ f j (Pi.mulSingle i x j)) = Pi.mulSingle i (f i x) := by
    funext j
    by_cases hj : j = i
    · subst j; simp
    · simp [Pi.mulSingle, Function.update_of_ne hj]
  rw [h]
  simp

@[simp] theorem liftFamily_comp_include {A : Type*} [CommRing A] [Algebra O A]
    (f : ∀ i, R i →ₐ[O] A) (i : ι) :
    (liftFamily R f).comp (includeFactor R i) = f i := by ext x; simp

/-- Uniqueness is proved using the actual finite tensor product's extensionality theorem. -/
theorem hom_ext {A : Type*} [CommRing A] [Algebra O A]
    {f g : Raw (O := O) R →ₐ[O] A}
    (h : ∀ i, f.comp (includeFactor R i) = g.comp (includeFactor R i)) : f = g :=
  PiTensorProduct.algHom_ext h

/-- The literal algebraic ideal generated by all local conditions. -/
def conditions (J : ∀ i, Ideal (R i)) : Ideal (Raw (O := O) R) :=
  iSup i, (J i).map (includeFactor R i).toRingHom

/-- The defining ideal also retains the base ideal, necessary for the empty family. -/
def definingIdeal (mO : Ideal O) (m : ∀ i, Ideal (R i)) : Ideal (Raw (O := O) R) :=
  mO.map (algebraMap O (Raw (O := O) R)) ⊔ conditions R m

variable (J : ∀ i, Ideal (R i))

/-- Apply the actual quotient map in every factor. -/
def quotientMap : Raw (O := O) R →ₐ[O] Raw (O := O) (fun i ↦ R i / J i) :=
  liftFamily R (fun i ↦ (includeFactor (fun i ↦ R i / J i) i).comp (Ideal.Quotient.mkₐ O (J i)))

@[simp] theorem quotientMap_include (i : ι) (x : R i) :
    quotientMap (O := O) R J (includeFactor R i x) =
      includeFactor (fun i ↦ R i / J i) i (Ideal.Quotient.mk (J i) x) := by
  simp [quotientMap]

/-- Each source condition vanishes in the actual tensor of quotients. -/
theorem conditions_le_ker : conditions (O := O) R J ≤ RingHom.ker (quotientMap (O := O) R J) := by
  rw [conditions, iSup_le_iff]
  intro i
  rw [Ideal.map_le_iff_le_comap]
  intro x hx
  change quotientMap (O := O) R J (includeFactor R i x) = 0
  rw [quotientMap_include, Ideal.Quotient.eq_zero_iff_mem.mpr hx, map_zero]

def quotientForward : (Raw (O := O) R / conditions (O := O) R J) →ₐ[O]
    Raw (O := O) (fun i ↦ R i / J i) :=
  Ideal.Quotient.liftₐ _ (quotientMap (O := O) R J) (conditions_le_ker R J)

def factorQuotient (i : ι) : (R i / J i) →ₐ[O] (Raw (O := O) R / conditions (O := O) R J) :=
  Ideal.Quotient.liftₐ (J i) ((Ideal.Quotient.mkₐ O (conditions (O := O) R J)).comp (includeFactor R i))
    (by
      intro x hx
      apply Ideal.Quotient.eq_zero_iff_mem.mpr
      exact (le_iSup (fun i ↦ (J i).map (includeFactor R i).toRingHom) i)
        (Ideal.mem_map_of_mem _ hx))

omit [Fintype ι] in
@[simp] theorem factorQuotient_mk (i : ι) (x : R i) :
    factorQuotient (O := O) R J i (Ideal.Quotient.mk (J i) x) =
      Ideal.Quotient.mk (conditions (O := O) R J) (includeFactor R i x) := rfl

def quotientBackward : Raw (O := O) (fun i ↦ R i / J i) →ₐ[O]
    (Raw (O := O) R / conditions (O := O) R J) :=
  liftFamily _ (factorQuotient (O := O) R J)

@[simp] theorem quotientForward_mk (x : Raw (O := O) R) :
    quotientForward (O := O) R J (Ideal.Quotient.mk _ x) = quotientMap (O := O) R J x := rfl

@[simp] theorem quotientBackward_include (i : ι) (x : R i / J i) :
    quotientBackward (O := O) R J (includeFactor (fun i ↦ R i / J i) i x) = factorQuotient (O := O) R J i x := by
  simp [quotientBackward]

theorem quotientForward_backward :
    (quotientForward (O := O) R J).comp (quotientBackward (O := O) R J) = AlgHom.id O _ := by
  apply hom_ext
  intro i
  ext x
  obtain ⟨x, rfl⟩ := Ideal.Quotient.mk_surjective x
  simp

theorem quotientBackward_forward :
    (quotientBackward (O := O) R J).comp (quotientForward (O := O) R J) = AlgHom.id O _ := by
  apply Ideal.Quotient.algHom_ext
  apply hom_ext
  intro i
  ext x
  simp

/-- Finite simultaneous tensor quotient, at the algebraic level. -/
def quotientEquiv : (Raw (O := O) R / conditions (O := O) R J) ≃ₐ[O]
    Raw (O := O) (fun i ↦ R i / J i) :=
  AlgEquiv.ofAlgHom (quotientForward (O := O) R J) (quotientBackward (O := O) R J)
    (quotientForward_backward R J) (quotientBackward_forward R J)

@[simp] theorem quotientEquiv_mk (x : Raw (O := O) R) :
    quotientEquiv (O := O) R J (Ideal.Quotient.mk _ x) = quotientMap (O := O) R J x := rfl

/-- The tensor-of-quotients map is an actual surjection. -/
theorem quotientMap_surjective : Function.Surjective (quotientMap (O := O) R J) := by
  intro y
  obtain ⟨x, hx⟩ := (quotientEquiv (O := O) R J).surjective y
  obtain ⟨z, rfl⟩ := Ideal.Quotient.mk_surjective x
  exact ⟨z, hx⟩

/-- Applying a coefficient quotient maps the topological defining ideal exactly. -/
theorem quotientMap_definingIdeal (mO : Ideal O) (m : ∀ i, Ideal (R i)) :
    (definingIdeal R mO m).map (quotientMap (O := O) R J).toRingHom =
      definingIdeal (fun i ↦ R i / J i) mO
        (fun i ↦ (m i).map (Ideal.Quotient.mk (J i))) := by
  simp only [definingIdeal, conditions, Ideal.map_sup, Ideal.map_iSup, Ideal.map_map]
  congr 1
  · congr 1
    ext r
    exact (quotientMap (O := O) R J).commutes r
  · apply congrArg iSup
    funext i
    congr 1
    ext x
    exact quotientMap_include R J i x

/-- The algebra equivalence carries the quotient's defining ideal to the actual target ideal. -/
theorem quotientEquiv_definingIdeal (mO : Ideal O) (m : ∀ i, Ideal (R i)) :
    ((definingIdeal R mO m).map (Ideal.Quotient.mk (conditions (O := O) R J))).map
      (quotientEquiv (O := O) R J).toRingEquiv.toRingHom =
      definingIdeal (fun i ↦ R i / J i) mO
        (fun i ↦ (m i).map (Ideal.Quotient.mk (J i))) := by
  rw [Ideal.map_map]
  exact quotientMap_definingIdeal R J mO m

end PaperXII.FiniteTensor
